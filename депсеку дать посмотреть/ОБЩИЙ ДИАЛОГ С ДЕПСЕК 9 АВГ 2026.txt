Отлично. Я вижу, что у тебя есть полная структура папок Crucix и ты готов к работе. Давай соберём всё воедино и создадим полноценный проект, который будет собирать данные из 400+ RSS-лент, складировать их в "корзину", и анализировать через AI.

Я предлагаю следующий план действий:

Создадим структуру для RSS-лент — папку feeds/ с файлом feeds.opml, куда мы поместим все 400+ адресов из твоего списка.

Напишем универсальный скрипт-парсер scripts/collect-feeds.mjs, который будет:

Читать feeds.opml

Парсить все RSS-ленты

Сохранять данные в /data/raw/ с ротацией по дням

Интегрируем модуль "Геополитика + AI" (уже есть в apis/sources/geopolitical-reports.mjs)

Создадим единую "корзину" — папку /data/raw/, куда все источники будут сбрасывать данные

Напишем скрипт-оркестратор, который будет раз в сутки:

Собирать все данные из /data/raw/

Отправлять их в локальный AI (Ollama) для фильтрации

Отправлять сжатый дайджест в DeepSeek через API

Сохранять финальный отчёт в /reports/

Давай начнём с самого важного — структуры RSS-лент.

Шаг 2. Создаём скрипт-сборщик для RSS-лент
Теперь создадим скрипт, который будет парсить все эти ленты и складывать их в "корзину" /data/raw/.

cat > "/home/ta8_/Рабочий стол/Crucix/scripts/collect-feeds.mjs" << 'EOF'
#!/usr/bin/env node

import { promises as fs } from 'fs';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';
import { createHash } from 'crypto';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = join(__dirname, '..');
const DATA_DIR = join(ROOT, 'data', 'raw');
const FEEDS_FILE = join(ROOT, 'data', 'feeds', 'feeds.opml');

// Убедимся, что папка существует
await fs.mkdir(DATA_DIR, { recursive: true });

// Парсинг OPML
function parseOpml(xml) {
  const feeds = [];
  const regex = /<outline[^>]*type="rss"[^>]*text="([^"]*)"[^>]*xmlUrl="([^"]*)"/g;
  let match;
  while ((match = regex.exec(xml)) !== null) {
    feeds.push({ name: match[1], url: match[2] });
  }
  return feeds;
}

// Парсинг RSS-ленты
async function fetchFeed(url) {
  try {
    const res = await fetch(url, {
      headers: { 'User-Agent': 'Crucix-Collector/1.0' },
      signal: AbortSignal.timeout(10000)
    });
    if (!res.ok) return [];
    const text = await res.text();

    // Простой парсинг RSS (без библиотек)
    const items = [];
    const itemRegex = /<item>([\s\S]*?)<\/item>/g;
    let itemMatch;
    while ((itemMatch = itemRegex.exec(text)) !== null) {
      const content = itemMatch[1];
      const title = (content.match(/<title>([\s\S]*?)<\/title>/) || [])[1] || 'Без заголовка';
      const link = (content.match(/<link>([\s\S]*?)<\/link>/) || [])[1] || '';
      const pubDate = (content.match(/<pubDate>([\s\S]*?)<\/pubDate>/) || [])[1] || '';
      const description = (content.match(/<description>([\s\S]*?)<\/description>/) || [])[1] || '';
      const id = createHash('md5').update(link).digest('hex');
      items.push({ id, title: title.trim(), link, pubDate, description: description.trim() });
    }
    return items;
  } catch (e) {
    console.error(`Ошибка при загрузке ${url}:`, e.message);
    return [];
  }
}

// Основная функция
async function collectAllFeeds() {
  console.log('[Collector] Чтение списка RSS-лент...');
  const xml = await fs.readFile(FEEDS_FILE, 'utf-8');
  const feeds = parseOpml(xml);
  console.log(`[Collector] Найдено ${feeds.length} лент`);

  const allItems = [];
  let processed = 0;

  for (const feed of feeds) {
    processed++;
    if (processed % 10 === 0) {
      console.log(`[Collector] Обработано ${processed}/${feeds.length}`);
    }
    const items = await fetchFeed(feed.url);
    for (const item of items) {
      allItems.push({
        ...item,
        source: feed.name,
        collectedAt: new Date().toISOString(),
        category: 'news'
      });
    }
    // Задержка, чтобы не перегружать серверы
    await new Promise(resolve => setTimeout(resolve, 200));
  }

  // Сохраняем в файл с ротацией по дням
  const today = new Date().toISOString().slice(0, 10);
  const outputFile = join(DATA_DIR, `feeds_${today}.json`);

  // Читаем существующий файл, если есть (для избежания дубликатов)
  let existing = [];
  try {
    const old = await fs.readFile(outputFile, 'utf-8');
    existing = JSON.parse(old);
  } catch (e) {
    // Файла нет — нормально
  }

  // Объединяем и удаляем дубликаты по id
  const combined = [...existing, ...allItems];
  const unique = new Map();
  for (const item of combined) {
    if (!unique.has(item.id)) {
      unique.set(item.id, item);
    }
  }

  const final = Array.from(unique.values());
  await fs.writeFile(outputFile, JSON.stringify(final, null, 2));
  console.log(`[Collector] Сохранено ${final.length} записей в ${outputFile}`);
}

// Запуск
collectAllFeeds().catch(console.error);

import { promises as fs } from 'fs';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = join(__dirname, '..');

const DAYS = 7; // Хранить 7 дней

async function cleanDir(dir) {
  try {
    const files = await fs.readdir(dir);
    const now = Date.now();
    let deleted = 0;
    for (const file of files) {
      const path = join(dir, file);
      const stat = await fs.stat(path);
      const age = (now - stat.mtimeMs) / (1000 * 60 * 60 * 24);
      if (age > DAYS) {
        await fs.unlink(path);
        deleted++;
      }
    }
    if (deleted > 0) {
      console.log(`[Clean] Удалено ${deleted} файлов из ${dir}`);
    }
  } catch (e) {
    // Папка может не существовать
  }
}

await cleanDir(DATA_DIR);
await cleanDir(REPORTS_DIR);
await cleanDir(AI_RAW_DIR);
console.log('[Clean] Готово.');

Все требуемые модули для Crucix в порядке целесообразности
Модуль №1 — Многоуровневый сборщик новостей и событий
Этот модуль должен агрегировать новости из всех доступных открытых источников одновременно: RSS-ленты, публичные Telegram-каналы, MAX, новостные API без ключей. Он не просто парсит один тип источников, а работает как единый шлюз, который принимает данные из десятков каналов и складывает их в общую корзину. Без этого модуля все остальные будут работать с урезанными данными. Он должен уметь работать через Tor для западных ресурсов и напрямую для российских, чтобы обходить блокировки.

Модуль №2 — Интеллектуальный фильтр шума на базе локального ИИ
Этот модуль получает сырые данные из корзины и прогоняет их через локальную модель (Ollama) для отсеивания мусора. Он должен уметь отличать реальные события от мелких происшествий, выделять аномалии и резюмировать тысячи новостей в 10-20 ключевых пунктов. Это критически важно, потому что без фильтрации анализировать сотни тысяч сообщений в день невозможно. Он работает полностью локально и не отправляет данные наружу.

Модуль №3 — Индекс глобальной напряжённости
Этот модуль вычисляет числовой показатель риска для каждой страны и для мира в целом на основе данных из всех остальных модулей. Он учитывает военные полёты, протесты, экономические санкции, передвижения флота, кибератаки и риторику лидеров. Алгоритм должен быть прозрачным и показывать, из каких компонентов складывается каждая цифра. Это главный инструмент для оценки приближения крупных конфликтов.

Модуль №4 — Сравнительный исторический анализ
Этот модуль сохраняет все ежедневные отчёты в векторную базу данных и позволяет сравнивать сегодняшние события с тем, что было неделю, месяц или год назад. Он отвечает на вопросы: «Это уже было?», «Как тогда развивались события?», «Чем текущая ситуация отличается от предыдущей эскалации?». Без истории невозможно понять тренды и отличить случайность от закономерности.

Модуль №5 — Слои критической инфраструктуры
Этот модуль добавляет на карту статические и динамические слои: подводные интернет-кабели, нефте- и газопроводы, стратегические проливы, военные базы, атомные электростанции, порты и аэродромы. Он показывает уязвимости цепочек поставок и позволяет быстро оценить, какие объекты окажутся под угрозой при эскалации в том или ином регионе.

Модуль №6 — Интеграция с российским мессенджером MAX
Этот модуль подключает сбор данных из MAX, который является государственным российским мессенджером и содержит официальные каналы госорганов и госкомпаний. В отличие от Telegram, MAX не блокируется в России и не требует VPN. Модуль должен уметь читать публичные каналы, извлекать сообщения и медиа-файлы, а также собирать статистику по постам.

Модуль №7 — Аналитические центры и прогнозные отчёты
Этот модуль собирает RSS-ленты ведущих мировых аналитических центров: RAND, CSIS, IISS, ISW, Chatham House, Carnegie Endowment, Brookings, Atlantic Council и других. Он не просто показывает ссылки, а парсит краткие содержания отчётов и добавляет их в общий поток данных для ИИ-анализа. Это даёт доступ к готовым экспертным оценкам, которые не появляются в обычных новостях.

Модуль №8 — Киберинтеллект и мониторинг угроз
Этот модуль подключает открытые ленты киберугроз: Feodo Tracker для отслеживания C2-серверов, AbuseIPDB для вредоносных IP-адресов, AlienVault OTX для индикаторов компрометации, CISA KEV для критических уязвимостей. Он должен показывать, какие страны и сектора находятся под атакой, и коррелировать киберактивность с геополитическими событиями.

Модуль №9 — Военная авиация с детектором аномалий
Этот модуль расширяет базовый трекинг самолётов (OpenSky) алгоритмом выявления аномалий: необычные маршруты, полёты без транспондеров, скопления военной авиации в необычных местах, нарушения воздушных границ. Он должен хранить историю полётов за 90 дней и показывать отклонения от нормы, а не просто текущие позиции.

Модуль №10 — Морской трекинг с детектором тёмных судов
Этот модуль отслеживает корабли через AIS и дополняет это алгоритмом выявления судов с выключенными транспондерами, сменой маршрута или нахождением в запретных зонах. Он должен показывать историю перемещений и подсвечивать подозрительную активность вблизи стратегических объектов, военных баз и подводных кабелей.

Модуль №11 — Спутниковый мониторинг и природные катастрофы
Этот модуль собирает данные со спутников: пожары через NASA FIRMS, землетрясения через USGS, наводнения и ураганы через GDACS, качество воздуха через OpenAQ, уровень радиации через Safecast. Он должен показывать не только текущие события, но и тренды — например, увеличение числа пожаров в определённом регионе может указывать на военные действия.

Модуль №12 — Экономический и финансовый дашборд
Этот модуль агрегирует данные с открытых биржевых API: Yahoo Finance для акций, CoinGecko для криптовалют, FRED для макроэкономики США, EIA для энергоносителей. Он должен показывать корреляцию между экономическими показателями и геополитическими событиями — например, как цена на нефть меняется при эскалации в определённом регионе.

Модуль №13 — Лента живых новостных потоков
Этот модуль встраивает в интерфейс прямые видео- и аудиопотоки с 25+ каналов: BBC, Al Jazeera, France 24, NHK, CGTN, RT и других. Он позволяет переключаться между каналами прямо в панели и слушать/смотреть новости в реальном времени, не открывая отдельные вкладки браузера.

Модуль №14 — Тематические «линзы» или варианты интерфейса
Этот модуль создаёт переключаемые профили интерфейса: «Геополитика», «Финансы», «Технологии», «Ресурсы», «Позитив». Каждый профиль показывает только релевантные слои и источники для конкретной темы. Это позволяет не захламлять интерфейс данными, которые не нужны в данный момент, и быстро переключаться между разными режимами анализа.

Модуль №15 — Многоязычный смысловой анализ
Этот модуль использует возможности AI для анализа новостей на языке оригинала, а не только в машинном переводе. Он должен понимать культурные и политические нюансы, различать формулировки в разных языковых версиях одного события и показывать, как одно и то же событие подаётся в разных информационных полях (российском, западном, китайском, арабском).

Модуль №16 — Симулятор сценариев «Что-если»
Этот модуль позволяет пользователю вручную изменять ключевые параметры — цену нефти, уровень военной активности, введение санкций, закрытие проливов — и наблюдать, как это влияет на другие показатели в модели. Он не даёт точных прогнозов, но показывает возможные цепочки последствий на основе исторических данных и текущих трендов.

Модуль №17 — Цифровой двойник пользователя
Этот модуль позволяет загружать в систему свои документы: контракты, списки активов, логистические маршруты, контакты в определённых регионах. AI-агент анализирует глобальные события через призму этих данных и подсвечивает только те новости, которые реально влияют на интересы пользователя. Это делает систему персонализированной, а не абстрактной.

Модуль №18 — Детектор информационной тишины
Этот модуль анализирует не только поток новостей, но и его отсутствие. Если в регионе, где обычно 100 новостей в день, вдруг становится 5, и при этом аномально растёт военная активность или движение кораблей — модуль выдаёт предупреждение о возможном скрытом событии. Он ищет «белый шум» и показывает, где информацию сознательно или случайно скрывают.

Модуль №19 — Кросс-корреляционный анализатор сигналов
Этот модуль ищет скрытые связи между разными типами данных: например, рост цен на пшеницу + увеличение военных учений + закрытие пролива = возможный продовольственный кризис. Он не просто показывает данные по отдельности, а вычисляет взаимосвязи и выдаёт комплексные сигналы тревоги или возможности.

Модуль №20 — Векторная память и RAG-поиск по истории
Этот модуль подключает локальную векторную базу данных (ChromaDB или Qdrant), куда сохраняются все ежедневные отчёты, аналитические сводки и ключевые выводы. Пользователь может задать вопрос на естественном языке и получить ответ, основанный на всей накопленной истории, а не только на текущем дне. Это превращает систему из простого дашборда в полноценную базу знаний.

Модуль №21 — Интеграция с открытыми AI-шлюзами
Этот модуль добавляет поддержку OpenRouter и других агрегаторов AI-моделей, чтобы пользователь мог выбирать не только конкретного провайдера, но и получать доступ к бесплатным или дешёвым моделям через единый API. Это снижает порог входа для тех, у кого нет возможности платить за DeepSeek или другие коммерческие API.

Модуль №22 — Экспорт данных и отчётов в разных форматах
Этот модуль добавляет кнопки для выгрузки собранных данных и аналитических отчётов в форматах JSON, CSV, PDF, HTML. Пользователь может сохранить текущий дайджест для архива, поделиться им или использовать в других системах. В текущей версии Crucix нет возможности сохранять результаты анализа, кроме как копировать текст вручную.

Модуль №23 — Автоматический планировщик задач внутри интерфейса
Этот модуль добавляет в интерфейс визуальный планировщик, где пользователь может настроить расписание сбора данных, фильтрации и генерации отчётов без использования внешнего cron. Можно задать разное расписание для разных тем: геополитику проверять каждый день, роботизацию — раз в неделю, а технологии — раз в месяц. Это делает систему удобной для не-технических пользователей.

Модуль №24 — Модуль оценки качества и доверия к источникам
Этот модуль автоматически оценивает каждый источник данных по шкале надёжности: государственное СМИ, независимое СМИ, аналитический центр, блог, анонимный канал. Он также сравнивает, как одно и то же событие освещается в разных группах источников, и показывает степень предвзятости. Это означает, что название снимка будет в формате:
