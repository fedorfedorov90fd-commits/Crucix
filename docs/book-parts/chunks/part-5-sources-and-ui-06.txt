<script>
const $ = id => document.getElementById(id);
const CONFIG = { dataUrl: '../runs/predictions/latest_forecast.json', refreshMs: 60000 };

const COLORS = [
  '#3b82f6', '#8b5cf6', '#ec4899', '#ef4444', '#f97316',
  '#eab308', '#22c55e', '#06b6d4', '#14b8a6', '#a855f7',
  '#f43f5e', '#84cc16', '#0ea5e9', '#6366f1', '#d946ef',
];

let DATA = null;

async function load() {
  try {
    const res = await fetch(CONFIG.dataUrl);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    DATA = await res.json();
    render();
  } catch (e) {
    console.error('Load error:', e);
  }
}

function render() {
  if (!DATA) return;
  const ad = DATA.attentionDynamics;
  if (!ad) {
    document.querySelectorAll('.empty').forEach(el => el.textContent = 'Нет данных об attention dynamics');
    return;
  }

  renderStats(ad);
  renderAllocation(ad);
  renderExplosive(ad);
  renderTimeline(ad);
  renderShifts(ad);
}

function renderStats(ad) {
  $('stat-topics').textContent = ad.totalTopics || 0;
  $('stat-total').textContent = (ad.totalAttention || 0).toFixed(1);
  $('stat-shifts').textContent = ad.shifts?.length || 0;
  $('stat-explosive').textContent = (ad.topExplosive || []).filter(e => e.willExplode).length;
}

function renderAllocation(ad) {
  const svg = $('allocation-chart');
  const w = svg.clientWidth || 600;
  const h = 400;
  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
  svg.innerHTML = '';

  const alloc = ad.attentionAllocation || {};
  const entries = Object.entries(alloc)
    .map(([id, data]) => ({ id, ...data }))
    .sort((a, b) => b.share - a.share)
    .slice(0, 15);

  if (entries.length === 0) {
    svg.innerHTML = `<text x="${w/2}" y="${h/2}" text-anchor="middle" fill="#5a6b8a" font-size="12">Нет данных</text>`;
    return;
  }

  const radius = Math.min(w, h) / 2 - 60;
  const cx = w / 2;
  const cy = h / 2;

  // Pie chart
  const pie = d3.pie().value(d => d.share).sort(null);
  const arc = d3.arc().innerRadius(radius * 0.5).outerRadius(radius);

  const g = d3.select(svg).append('g')
    .attr('transform', `translate(${cx},${cy})`);

  g.selectAll('.alloc-slice')
    .data(pie(entries))
    .enter()
    .append('path')
    .attr('class', 'alloc-slice')
    .attr('d', arc)
    .attr('fill', (d, i) => COLORS[i % COLORS.length])
    .on('mouseenter', (event, d) => {
      showTooltip(event, `
        <strong>${d.data.id}</strong><br>
        Доля: ${(d.data.share * 100).toFixed(1)}%<br>
        Внимание: ${d.data.attention.toFixed(2)}<br>
        Momentum: ${d.data.momentum >= 0 ? '+' : ''}${d.data.momentum.toFixed(2)}<br>
        Velocity: ${d.data.velocity.toFixed(2)}
      `);
    })
    .on('mousemove', moveTooltip)
    .on('mouseleave', hideTooltip);

  // Labels для топ-5
  const top5 = pie(entries.slice(0, 5));
  const labelArc = d3.arc().innerRadius(radius * 0.75).outerRadius(radius * 0.75);

  g.selectAll('.alloc-label')
    .data(top5)
    .enter()
    .append('text')
    .attr('class', 'alloc-label')
    .attr('transform', d => `translate(${labelArc.centroid(d)})`)
    .attr('text-anchor', 'middle')
    .text(d => d.data.id.substring(0, 10));

  g.selectAll('.alloc-value')
    .data(top5)
    .enter()
    .append('text')
    .attr('class', 'alloc-value')
    .attr('transform', d => {
      const c = labelArc.centroid(d);
      return `translate(${c[0]},${c[1] + 12})`;
    })
    .attr('text-anchor', 'middle')
    .text(d => `${(d.data.share * 100).toFixed(0)}%`);
}

function renderExplosive(ad) {
  const list = $('explosive-list');
  const explosive = (ad.topExplosive || []).slice(0, 10);

  if (explosive.length === 0) {
    list.innerHTML = '<div class="empty">Нет данных</div>';
    return;
  }

  list.innerHTML = explosive.map(e => {
    const color = e.willExplode ? '#ef4444' :
                  e.explosiveness > 0.5 ? '#f97316' :
                  e.explosiveness > 0.2 ? '#eab308' : '#22c55e';
    return `
      <div class="explosive-item ${e.willExplode ? 'exploding' : ''}">
        <div class="explosive-header">
          <span class="explosive-topic">${e.topic}</span>
          <span class="explosive-score" style="color: ${color};">
            ${e.explosiveness.toFixed(2)}
          </span>
        </div>
        <div class="explosive-meta">
          attention: ${e.currentAttention.toFixed(1)} → ${e.predictedAttention.toFixed(1)}
          · velocity: ${e.velocity.toFixed(2)}
          ${e.willExplode ? ' · ⚡ EXPLODING' : ''}
        </div>
      </div>
    `;
  }).join('');
}

function renderTimeline(ad) {
  const svg = $('timeline-chart');
  const w = svg.clientWidth || 1200;
  const h = 350;
  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
  svg.innerHTML = '';

  // Извлекаем историю из топ-10 тем
  const topics = (ad.topExplosive || []).slice(0, 10).map(e => e.topic);
  if (topics.length === 0) {
    svg.innerHTML = `<text x="${w/2}" y="${h/2}" text-anchor="middle" fill="#5a6b8a" font-size="12">Нет данных</text>`;
    return;
  }

  // Проверяем, есть ли история в данных
  const hasHistory = ad.attentionAllocation && Object.values(ad.attentionAllocation).some(v => v.history);

  if (!hasHistory) {
    // Fallback: рисуем bars для текущего attention
    const barData = topics.map(t => ({
      topic: t,
      attention: ad.attentionAllocation[t]?.attention || 0,
    }));

    const margin = { top: 20, right: 30, bottom: 60, left: 60 };
    const pw = w - margin.left - margin.right;
    const ph = h - margin.top - margin.bottom;

    const maxA = Math.max(...barData.map(d => d.attention), 1);
    const xScale = d3.scaleBand().domain(topics).range([0, pw]).padding(0.2);
    const yScale = d3.scaleLinear().domain([0, maxA]).range([ph, 0]);

    const g = d3.select(svg).append('g').attr('transform', `translate(${margin.left},${margin.top})`);

    g.selectAll('.bar')
      .data(barData)
      .enter()
      .append('rect')
      .attr('x', d => xScale(d.topic))
      .attr('y', d => yScale(d.attention))
      .attr('width', xScale.bandwidth())
      .attr('height', d => ph - yScale(d.attention))
      .attr('fill', (d, i) => COLORS[i % COLORS.length])
      .attr('rx', 4);

    // Labels
    g.selectAll('.tl-label')
      .data(barData)
      .enter()
      .append('text')
      .attr('class', 'tl-axis')
      .attr('x', d => xScale(d.topic) + xScale.bandwidth() / 2)
      .attr('y', ph + 20)
      .attr('text-anchor', 'middle')
      .attr('font-size', '10')
      .text(d => d.topic.substring(0, 10));

    g.selectAll('.tl-value')
      .data(barData)
      .enter()
      .append('text')
      .attr('x', d => xScale(d.topic) + xScale.bandwidth() / 2)
      .attr('y', d => yScale(d.attention) - 6)
      .attr('text-anchor', 'middle')
      .attr('font-family', 'var(--mono)')
      .attr('font-size', '10')
      .attr('fill', 'var(--text)')
      .text(d => d.attention.toFixed(1));
  }
}

function renderShifts(ad) {
  const shifts = ad.shifts || [];
  const list = $('shifts-list');

  if (shifts.length === 0) {
    list.innerHTML = '<div class="empty">Нет обнаруженных сдвигов</div>';
    return;
  }

  list.innerHTML = shifts.map(s => `
    <div class="shift-item">
      <div class="shift-header">
        <span>
          <span class="shift-from">${s.from}</span>
          <span style="color: var(--text-faint);"> → </span>
          <span class="shift-to">${s.to}</span>
        </span>
        <span class="shift-magnitude">${s.magnitude.toFixed(2)}</span>
      </div>
      <div class="shift-desc">${s.interpretation}</div>
    </div>
  `).join('');
}

function showTooltip(e, html) {
  const tt = $('tooltip');
  tt.innerHTML = html;
  tt.classList.add('visible');
  moveTooltip(e);
}
function moveTooltip(e) {
  const tt = $('tooltip');
  tt.style.left = `${Math.min(e.clientX + 14, window.innerWidth - tt.offsetWidth - 10)}px`;
  tt.style.top = `${Math.min(e.clientY + 14, window.innerHeight - tt.offsetHeight - 10)}px`;
}
function hideTooltip() { $('tooltip').classList.remove('visible'); }

load();
setInterval(load, CONFIG.refreshMs);

let resizeTimer;
window.addEventListener('resize', () => {
  clearTimeout(resizeTimer);
  resizeTimer = setTimeout(render, 300);
});
</script>

</body>

<script>
const $ = id => document.getElementById(id);
const CONFIG = { dataUrl: '../runs/predictions/latest_forecast.json', refreshMs: 60000 };

const STRATEGY_COLORS = {
  escalate: '#ef4444',
  hold: '#eab308',
  deescalate: '#22c55e',
  deceive: '#8b5cf6',
  forecast_high: '#ef4444',
  forecast_low: '#22c55e',
  withhold: '#64748b',
};

const OPPONENT_COLORS = {
  russia: '#ef4444',
  china: '#f97316',
  usa: '#3b82f6',
  eu: '#22c55e',
  iran: '#8b5cf6',
  nato: '#06b6d4',
};

let DATA = null;

async function load() {
  try {
    const res = await fetch(CONFIG.dataUrl);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    DATA = await res.json();
    render();
  } catch (e) {
    console.error('Load error:', e);
  }
}

function render() {
  if (!DATA) return;
  const ace = DATA.adversarialCoEvolution;
  if (!ace) {
    document.querySelectorAll('.empty').forEach(el => el.textContent = 'Нет данных о co-evolution');
    return;
  }

  renderKeyInsight(ace);
  renderOpponents(ace);
  renderHistory(ace);
}

function renderKeyInsight(ace) {
  const el = $('key-insight');
  const adapting = (ace.adaptations || []).filter(a => a.adaptation?.adapted);

  if (adapting.length > 0) {
    el.className = 'key-insight';
    el.innerHTML = `
      ⚠ <strong>${adapting.length}</strong> противник(ов) адаптируется к нашим прогнозам<br>
      <span style="font-size: 12px; font-weight: 400; margin-top: 8px; display: block;">
        ${adapting.map(a => `${a.opponent.toUpperCase()} (KL=${a.adaptation.klDivergence.toFixed(2)})`).join(', ')}
      </span>
    `;
  } else {
    el.className = 'key-insight safe';
    el.innerHTML = `
      ✓ Противники не демонстрируют явной адаптации<br>
      <span style="font-size: 12px; font-weight: 400; margin-top: 8px; display: block;">
        Текущая стратегия прогнозирования оптимальна
      </span>
    `;
  }
}

function renderOpponents(ace) {
  const grid = $('opponents-grid');
  const adaptations = ace.adaptations || [];

  $('opponents-count').textContent = `${adaptations.length}`;

  if (adaptations.length === 0) {
    grid.innerHTML = '<div class="empty">Нет моделей противников</div>';
    return;
  }

  grid.innerHTML = adaptations.map(a => {
    const oppColor = OPPONENT_COLORS[a.opponent] || '#64748b';
    const isAdapting = a.adaptation?.adapted;
    const distribution = a.distribution || {};

    const strategyRows = Object.entries(distribution)
      .sort((x, y) => y[1] - x[1])
      .map(([strategy, prob]) => {
        const color = STRATEGY_COLORS[strategy] || '#64748b';
        return `
          <div class="strategy-row">
            <span class="strategy-name">${strategy}</span>
            <div class="strategy-track">
              <div class="strategy-fill" style="width: ${prob * 100}%; background: ${color};"></div>
            </div>
            <span class="strategy-value" style="color: ${color};">${(prob * 100).toFixed(1)}%</span>
          </div>
        `;
      }).join('');

    return `
      <div class="opponent-card ${isAdapting ? 'adapting' : ''}" style="border-left-color: ${oppColor};">
        <div class="opponent-header">
          <span class="opponent-name" style="color: ${oppColor};">${a.opponent}</span>
          <span class="opponent-adapt-badge ${isAdapting ? 'yes' : 'no'}">
            ${isAdapting ? '⚠ Адаптируется' : '✓ Стабилен'}
          </span>
        </div>
        <div class="strategy-distribution">
          ${strategyRows}
        </div>
        <div class="predicted-next">
          <div class="predicted-next-label">Прогноз следующей стратегии</div>
          <div class="predicted-next-value" style="color: ${STRATEGY_COLORS[a.predictedStrategy] || oppColor};">
            ${a.predictedStrategy} (${(a.predictedProbability * 100).toFixed(1)}%)
          </div>
        </div>
        ${a.adaptation?.klDivergence !== undefined ? `
          <div class="kl-info">
            KL divergence: ${a.adaptation.klDivergence.toFixed(3)}
            ${isAdapting ? '(> 0.3 = адаптация)' : ''}
          </div>
        ` : ''}
      </div>
    `;
  }).join('');
}

function renderHistory(ace) {
  const list = $('history-list');
  const history = ace.evolutionHistory || [];

  if (history.length === 0) {
    list.innerHTML = '<div class="empty">Нет истории симуляции</div>';
    return;
  }

  list.innerHTML = history.reverse().slice(0, 20).map((h, idx) => {
    const actionsHtml = (h.actions || []).map(a => `
      <div style="padding: 6px 8px; background: var(--surface3); border-radius: 6px; margin-bottom: 4px;">
        <div class="history-header">
          <span style="color: ${OPPONENT_COLORS[a.opponent] || 'var(--text)'};">${a.opponent}</span>
          <span>наблюдалось: <span class="history-observed">${a.observed}</span></span>
        </div>
        <div style="display: flex; justify-content: space-between; font-size: 10px; color: var(--text-faint);">
          <span>прогноз: <span class="history-predicted">${a.predicted}</span></span>
          <span>${a.adaptation?.adapted ? '⚠ адаптация' : 'ok'}</span>
        </div>
      </div>
    `).join('');

    return `
      <div class="history-item">
        <div class="history-header">
          <span style="color: var(--accent2); font-weight: 700;">Раунд #${h.round || idx + 1}</span>
          <span class="history-meta">${actionsHtml ? '' : '(нет действий)'}</span>
        </div>
        ${actionsHtml}
      </div>
    `;
  }).join('');
}

function showTooltip(e, html) {
  const tt = $('tooltip');
  tt.innerHTML = html;
  tt.classList.add('visible');
  moveTooltip(e);
}
function moveTooltip(e) {
  const tt = $('tooltip');
  tt.style.left = `${Math.min(e.clientX + 14, window.innerWidth - tt.offsetWidth - 10)}px`;
  tt.style.top = `${Math.min(e.clientY + 14, window.innerHeight - tt.offsetHeight - 10)}px`;
}
function hideTooltip() { $('tooltip').classList.remove('visible'); }

load();
setInterval(load, CONFIG.refreshMs);

let resizeTimer;
window.addEventListener('resize', () => {
  clearTimeout(resizeTimer);
  resizeTimer = setTimeout(render, 300);
});
</script>
