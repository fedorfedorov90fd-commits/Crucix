const CONFIG = {
  apiBase: window.location.origin,
  wsUrl: `ws://${window.location.host}/agent`,
  pollIntervalMs: 5000,
  useWebSocket: false, // переключается при успешном подключении
};

const TOOLS = [
  { name: 'engine_full', category: 'base' },
  { name: 'regime_shift', category: 'base' },
  { name: 'explainability', category: 'base' },
  { name: 'neural_causal_discovery', category: 'v6' },

  { name: 'continuous_causal', category: 'v7' },
  { name: 'simulation_engine', category: 'v7' },
];

// ═══════════════════════════════════════════════════

let ws = null;
let isProcessing = false;
let sessionCounter = 0;

// ═══════════════════════════════════════════════════
// WEB SOCKET
// ═══════════════════════════════════════════════════

function connectWS() {
  try {
    ws = new WebSocket(CONFIG.wsUrl);
    ws.onopen = () => {
      CONFIG.useWebSocket = true;
      updateStatus('ws', 'ok', 'WS');
      console.log('[agent] WebSocket подключён');
    };
    ws.onmessage = (ev) => {
      try {
        const msg = JSON.parse(ev.data);
        if (msg.type === 'agent_response') {
          renderAgentResponse(msg.data);
          isProcessing = false;
          updateSendButton();
        } else if (msg.type === 'agent_status') {
          updateSystemStatus(msg.data);
        }
      } catch (e) {
        console.error('[agent] parse error:', e);
      }
    };
    ws.onclose = () => {
      CONFIG.useWebSocket = false;
      updateStatus('ws', 'warn', 'WS off');
      // Переподключение через 5 секунд
      setTimeout(connectWS, 5000);
    };
    ws.onerror = () => {
      CONFIG.useWebSocket = false;
    };
  } catch (e) {
    CONFIG.useWebSocket = false;
  }
}

// ═══════════════════════════════════════════════════
// HTTP API
// ═══════════════════════════════════════════════════

async function callAgentHTTP(query) {
  const sessionId = `ui_${Date.now()}_${++sessionCounter}`;
  try {
    const resp = await fetch(`${CONFIG.apiBase}/api/agent/query`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ query, sessionId }),
    });
    if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
    return await resp.json();
  } catch (e) {
    return {
      error: `Не удалось вызвать agent: ${e.message}. Проверь что сервер /api/agent/query запущен.`,
      sessionId,
    };
  }
}

// ═══════════════════════════════════════════════════
// UI
// ═══════════════════════════════════════════════════

function updateStatus(name, state, label) {
  const dot = document.getElementById(`${name}-status`);
  const lbl = document.getElementById(`${name}-label`);
  if (dot) {
    dot.className = 'dot ' + state;
  }
  if (lbl && label) lbl.textContent = label;
}

function updateSystemStatus(data) {
  if (!data) return;
  const set = (id, v, cls) => {
    const el = document.getElementById(id);
    if (!el) return;
    el.textContent = v;
    el.className = 'v' + (cls ? ' ' + cls : '');
  };
  if (data.llm !== undefined) set('ss-llm', data.llm ? 'online' : 'offline', data.llm ? 'ok' : 'err');
  if (data.tools !== undefined) set('ss-tools', String(data.tools), data.tools > 0 ? 'ok' : 'err');
  if (data.history !== undefined) set('ss-history', String(data.history), data.history >= 20 ? 'ok' : 'warn');
  if (data.snapshot !== undefined) set('ss-snapshot', data.snapshot ? 'found' : 'missing', data.snapshot ? 'ok' : 'warn');
}

function appendMessage(role, content, meta) {
  const msg = document.createElement('div');
  msg.className = `msg ${role}`;
  if (meta) {
    const metaEl = document.createElement('div');
    metaEl.className = 'meta';
    for (const [k, v] of Object.entries(meta)) {
      const span = document.createElement('span');
      span.className = `badge ${v}`;
      span.textContent = k;
      metaEl.appendChild(span);
    }
    msg.appendChild(metaEl);
  }
  const textEl = document.createElement('div');
  textEl.textContent = content;
  msg.appendChild(textEl);
  document.getElementById('messages').appendChild(msg);
  msg.scrollIntoView({ behavior: 'smooth', block: 'end' });
  return msg;
}

function appendThinking() {
  const el = document.createElement('div');
  el.className = 'thinking';
  el.innerHTML = 'Agent <span class="dots"></span>';
  el.id = 'thinking-indicator';
  document.getElementById('messages').appendChild(el);
  el.scrollIntoView({ behavior: 'smooth', block: 'end' });
  return el;
}

function removeThinking() {
  const el = document.getElementById('thinking-indicator');
  if (el) el.remove();
}

function renderAgentResponse(data) {
  removeThinking();

  if (data.error) {
    appendMessage('agent', `❌ ${data.error}`, { error: 'err' });
    return;
  }

  const meta = {
    mode: data.mode || 'unknown',
    confidence: data.answer?.confidence || data.confidence || 'unknown',
    source: data.answer?.source || 'template',
  };

  const text = data.answer?.text || data.answer || data.text || '(нет ответа)';
  const msg = appendMessage('agent', text, meta);

  // Раскрываемые детали (план + executions)
  if (data.plan || data.executions) {
    const details = document.createElement('details');
    const summary = document.createElement('summary');
    summary.textContent = `Детали (${data.plan?.length || 0} шагов, ${data.executions?.length || 0} вызовов)`;
    details.appendChild(summary);

    const pre = document.createElement('pre');
    pre.textContent = JSON.stringify({
      mode: data.mode,
      reasoning: data.reasoning,
      plan: data.plan,
      executions: data.executions?.map(e => ({
        tool: e.tool,
        args: e.args,
        ok: e.ok,
        error: e.error || null,
        elapsedMs: e.elapsedMs,
        resultKeys: e.result ? Object.keys(e.result) : null,
      })),
    }, null, 2);
    details.appendChild(pre);
    msg.appendChild(details);
  }
}

function updateSendButton() {
  const btn = document.getElementById('send');
  btn.disabled = isProcessing;
  btn.textContent = isProcessing ? 'Обработка...' : 'Отправить';
}

// ═══════════════════════════════════════════════════
// ОТПРАВКА
// ═══════════════════════════════════════════════════

async function sendQuery() {
  const input = document.getElementById('input');
  const query = input.value.trim();
  if (!query || isProcessing) return;

  appendMessage('user', query);
  input.value = '';
  isProcessing = true;
  updateSendButton();
  appendThinking();

  if (CONFIG.useWebSocket && ws && ws.readyState === WebSocket.OPEN) {
    ws.send(JSON.stringify({ type: 'query', query }));
  } else {
    const result = await callAgentHTTP(query);
    renderAgentResponse(result);
    isProcessing = false;
    updateSendButton();
  }
}

// ═══════════════════════════════════════════════════
// ИНИЦИАЛИЗАЦИЯ
// ═══════════════════════════════════════════════════

function renderToolList() {
  const container = document.getElementById('tool-list');
  container.innerHTML = '';
  for (const tool of TOOLS) {
    const el = document.createElement('div');
    el.className = 'tool';
    el.innerHTML = `<span class="cat ${tool.category}">${tool.category}</span>${tool.name}`;
    container.appendChild(el);
  }
}

async function pollStatus() {
  try {
    const resp = await fetch(`${CONFIG.apiBase}/api/agent/status`);
    if (!resp.ok) throw new Error('status failed');
    const data = await resp.json();
    updateSystemStatus(data);
    updateStatus('llm', data.llm ? 'ok' : 'err', data.llm ? 'LLM online' : 'LLM off');
    updateStatus('history', data.history >= 20 ? 'ok' : 'warn', `History: ${data.history}`);
    updateStatus('snapshot', data.snapshot ? 'ok' : 'warn', data.snapshot ? 'Snapshot found' : 'No snapshot');
  } catch (e) {
    updateStatus('llm', 'warn', 'API off');
    updateStatus('history', 'warn', 'API off');
    updateStatus('snapshot', 'warn', 'API off');
  }
}

function init() {
  renderToolList();
  pollStatus();
  setInterval(pollStatus, CONFIG.pollIntervalMs);
  connectWS();

  const input = document.getElementById('input');
  input.addEventListener('keydown', (e) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      sendQuery();
    }
  });

  document.getElementById('send').addEventListener('click', sendQuery);

  document.querySelectorAll('.quick-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      input.value = btn.dataset.q;
      sendQuery();
    });
  });

  // Автовысота textarea
  input.addEventListener('input', () => {
    input.style.height = 'auto';
    input.style.height = Math.min(150, input.scrollHeight) + 'px';
  });
}

document.addEventListener('DOMContentLoaded', init);

<script>
const $ = id => document.getElementById(id);

async function load() {
try {
const res = await fetch('../runs/predictions/latest_forecast.json');
if (!res.ok) throw new Error('HTTP ' + res.status);
const data = await res.json();
render(data);
} catch (e) {
document.querySelectorAll('.empty').forEach(el => el.textContent = 'Нет данных: ' + e.message);
}
}

function render(data) {
if (data.timestamp) $('last-update').textContent = new Date(data.timestamp).toLocaleTimeString('ru-RU');
if (data.metaEnsemble?.currentRegime) $('regime').textContent = 'regime: ' + data.metaEnsemble.currentRegime;

// Stats
$('stat-causal').textContent = data.combinedCausalSignal?.type || '—';
$('stat-warfare').textContent = data.narrativeWarfare?.campaignsDetected || 0;
$('stat-resources').textContent = data.resourceExhaustion?.overallStatus || '—';
$('stat-scenarios').textContent = data.scenarios?.scenarioCount || 0;

// Temporal
if (data.temporalCausal) {
const tc = data.temporalCausal;
const shifts = tc.shiftDetections || [];
$('temporal').innerHTML = '<div style="font-size:11px;color:var(--text-dim);margin-bottom:8px">Событий: ' + (tc.totalEvents || 0) + ' · Сдвигов: ' + shifts.length + '</div>' +
(shifts.length > 0 ? shifts.slice(0, 5).map(s =>
'<div class="row" style="border-left-color:' + (s.type === 'accelerating' ? '#ef4444' : '#06b6d4') + '">' +
'<div class="row-header"><span class="row-name">' + s.from + ' → ' + s.to + '</span>' +
'<span class="row-value" style="color:' + (s.type === 'accelerating' ? '#ef4444' : '#06b6d4') + '">' + s.shiftPct.toFixed(0) + '%</span></div>' +
'<div class="row-meta">' + s.type + ' · z=' + s.zScore.toFixed(2) + '</div></div>'
).join('') : '<div class="empty">Нет сдвигов</div>');
}

// Multi-layer
if (data.multiLayerCausal) {
const ml = data.multiLayerCausal;
const prop = ml.propagation || {};
$('multilayer').innerHTML = '<div style="font-size:11px;color:var(--text-dim);margin-bottom:8px">Узлов: ' + (ml.graphStats?.nodeCount || 0) + ' · Кросс-слойных: ' + (ml.graphStats?.crossLayerEdgeCount || 0) + '</div>' +
'<div style="font-size:11px;color:var(--text-dim)">Propagation updates: ' + (prop.totalUpdates || 0) + ' (cross-layer: ' + (prop.crossLayerUpdates || 0) + ')</div>';
}

// Warfare
if (data.narrativeWarfare) {
const nw = data.narrativeWarfare;
$('warfare').innerHTML = '<div style="font-size:13px;font-weight:700;color:var(--accent3);margin-bottom:8px">' + (nw.assessment || '—') + '</div>' +
'<div style="font-size:11px;color:var(--text-dim)">Кампаний: ' + (nw.campaignsDetected || 0) + ' · Угроза: ' + ((nw.threatLevel || 0) * 100).toFixed(0) + '%</div>';
}

// Resources
if (data.resourceExhaustion) {
const re = data.resourceExhaustion;
$('resources').innerHTML = '<div style="font-size:13px;font-weight:700;margin-bottom:8px">' + (re.overallStatus || '—') + '</div>' +
'<div style="font-size:11px;color:var(--text-dim)">Давление: ' + ((re.overallPressure || 0) * 100).toFixed(0) + '%</div>' +
(re.interpretation ? '<div style="font-size:10px;color:var(--text-faint);margin-top:6px">' + re.interpretation + '</div>' : '');
}

// Meta
if (data.metaEnsemble) {
const me = data.metaEnsemble;
const weights = Object.entries(me.modelWeights || {}).sort((a, b) => b[1] - a[1]).slice(0, 5);
$('meta').innerHTML = '<div style="font-size:13px;font-weight:700;color:var(--accent);margin-bottom:8px">Режим: ' + (me.currentRegime || '—') + '</div>' +
weights.map(([name, w]) =>
'<div style="display:flex;justify-content:space-between;padding:4px 0;font-size:11px">' +
'<span>' + name + '</span><span style="font-family:var(--mono);color:var(--accent)">' + (w * 100).toFixed(1) + '%</span></div>'
).join('');
}

// Scenarios
if (data.scenarios?.scenarios) {
$('scenarios').innerHTML = data.scenarios.scenarios.slice(0, 6).map(sc =>
'<div class="row">' +
'<div class="row-header"><span class="row-name">' + sc.name + '</span>' +
'<span class="row-value">' + ((sc.probability || 0) * 100).toFixed(0) + '%</span></div>' +
'<div class="row-meta">' + sc.severity + ' · ' + (sc.horizonHours || '?') + 'ч</div></div>'
).join('');
}
}

load();
setInterval(load, 60000);
</script>
