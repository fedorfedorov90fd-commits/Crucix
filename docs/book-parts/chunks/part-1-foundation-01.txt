### v4 -> v5 (текущая версия)

Пять новых модулей прогностического ядра:

- **Causal Hypergraph Discovery** (`apis/predict/hypergraph_discovery.mjs`) — автоматический поиск N-арных причинных связей через G²-тест и mutual information с permutation testing. Не требует ручной разметки гиперрёбер.
- **Multi-scale Attention** (`apis/predict/multiscale_attention.mjs`) — параллельное моделирование коллективного внимания на трёх временных горизонтах (fast/medium/slow с half-life 2/12/72 часа). Дивергенция между масштабами = шум, согласие = устойчивый тренд.
- **Opponent Modeling PPO** (`apis/predict/opponent_ppo.mjs`) — обучение модели противника через Proximal Policy Optimization. Actor-critic с GAE, PPO-clip, entropy regularization. Полный Adam с раздельными bias correction для actor и critic.
- **Federated Hypergraph** (`apis/predict/federated_hypergraph.mjs`) — распределённое обучение гиперграфов между инстансами через FedAvg с дифференциальной приватностью (Laplace + Gaussian noise).
- **Bayesian Causal Discovery** (`apis/predict/bayesian_causal.mjs`) — MCMC по пространству DAG с BDeu score, posterior edge probabilities, Markov blankets. Не точечная оценка, а распределение по структурам.

Плюс интеграционный патч:

- **Engine Integration Patch** (`apis/predict/engine_integration_patch.mjs`) — подключает 21 модуль (11 научных + 4 нейросетевых + 4 объяснимости + 3 трансляции) к основному конвейеру через четыре фазы: N (научный синтез), O (нейросетевой слой), P (объяснимость), Q (трансляция). Экспортирует `runNewModules` и `runForecastPipelineExtended`.

Единый дашборд:

- **Unified Cockpit** (`dashboard/cockpit.html`) — единая точка управления всеми дашбордами: 8 mission-карточек, глобальный status bar, live feed, command palette (Ctrl+K), 13-layer health monitoring, alert banner, sticky sidebars.

Тестирование:

Пишет два отчёта: `runs/mutation/report.json` (машиночитаемый) и `runs/mutation/report.md` (Markdown-дайджест). Ведёт регрессионный трекинг: сравнивает score с предыдущим запуском, отмечает тренды `^` (улучшение), `v` (деградация), `=` (без изменений), `*` (первый запуск).

function dot(a, b) { let s = 0; for (let i = 0; i < a.length; i++) s += a[i] * b[i]; return s; }
function gradientDescent({ f, grad, x0, lr = 0.01, iterations = 1000, tol = 1e-8, callback = null }) { let x = [...x0]; let fx = f(x); const history = [{ x: [...x], f: fx }]; for (let i = 0; i < iterations; i++) { const g = grad(x); const xNew = x.map((xi, j) => xi - lr * g[j]); const fNew = f(xNew); if (callback) callback({ iteration: i, x: xNew, f: fNew, grad: g }); if (Math.abs(fNew - fx) < tol) { x = xNew; fx = fNew; break; } x = xNew; fx = fNew; history.push({ x: [...x], f: fx }); } return { x, f: fx, iterations: history.length, history }; }
function adam({ f, grad, x0, lr = 0.001, beta1 = 0.9, beta2 = 0.999, eps = 1e-8, iterations = 1000, tol = 1e-8 }) { const n = x0.length; let x = [...x0]; const m = new Array(n).fill(0); const v = new Array(n).fill(0); const history = []; for (let t = 1; t <= iterations; t++) { const g = grad(x); for (let i = 0; i < n; i++) { m[i] = beta1 * m[i] + (1 - beta1) * g[i]; v[i] = beta2 * v[i] + (1 - beta2) * g[i] * g[i]; } const mHat = m.map(mi => mi / (1 - Math.pow(beta1, t))); const vHat = v.map(vi => vi / (1 - Math.pow(beta2, t))); for (let i = 0; i < n; i++) x[i] -= lr * mHat[i] / (Math.sqrt(vHat[i]) + eps); const fx = f(x); history.push({ iteration: t, f: fx, x: [...x] }); if (history.length > 1 && Math.abs(history[history.length - 1].f - history[history.length - 2].f) < tol) break; } return { x, f: f(x), iterations: history.length, history }; }
function rmsprop({ f, grad, x0, lr = 0.01, rho = 0.9, eps = 1e-8, iterations = 1000, tol = 1e-8 }) { const n = x0.length; let x = [...x0]; const cache = new Array(n).fill(0); const history = []; for (let t = 1; t <= iterations; t++) { const g = grad(x); for (let i = 0; i < n; i++) { cache[i] = rho * cache[i] + (1 - rho) * g[i] * g[i]; x[i] -= lr * g[i] / (Math.sqrt(cache[i]) + eps); } const fx = f(x); history.push({ iteration: t, f: fx }); if (history.length > 1 && Math.abs(history[history.length - 1].f - history[history.length - 2].f) < tol) break; } return { x, f: f(x), iterations: history.length, history }; }
function nelderMead({ f, x0, iterations = 1000, tol = 1e-8, alpha = 1, gamma = 2, rho = 0.5, sigma = 0.5 }) { const n = x0.length; let simplex = [x0.slice()]; for (let i = 0; i < n; i++) { const p = x0.slice(); p[i] += p[i] === 0 ? 0.00025 : p[i] * 0.05; simplex.push(p); } const fVals = simplex.map(f); let history = []; for (let iter = 0; iter < iterations; iter++) { const order = fVals.map((v, i) => ({ v, i })).sort((a, b) => a.v - b.v); simplex = order.map(o => simplex[o.i]); fVals.length = 0; fVals.push(...order.map(o => o.v)); const spread = Math.max(...fVals) - Math.min(...fVals); history.push({ iteration: iter, fBest: fVals[0], spread }); if (spread < tol) break; const centroid = new Array(n).fill(0); for (let i = 0; i < n; i++) for (let j = 0; j < n; j++) centroid[j] += simplex[i][j] / n; const xr = centroid.map((c, j) => c + alpha * (c - simplex[n][j])); const fr = f(xr); if (fr < fVals[0]) { const xe = centroid.map((c, j) => c + gamma * (xr[j] - c)); const fe = f(xe); if (fe < fr) { simplex[n] = xe; fVals[n] = fe; } else { simplex[n] = xr; fVals[n] = fr; } } else if (fr < fVals[n - 1]) { simplex[n] = xr; fVals[n] = fr; } else { const xc = centroid.map((c, j) => c + rho * (simplex[n][j] - c)); const fc = f(xc); if (fc < fVals[n]) { simplex[n] = xc; fVals[n] = fc; } else { for (let i = 1; i <= n; i++) { simplex[i] = simplex[i].map((v, j) => simplex[0][j] + sigma * (v - simplex[0][j])); fVals[i] = f(simplex[i]); } } } } const order = fVals.map((v, i) => ({ v, i })).sort((a, b) => a.v - b.v); return { x: simplex[order[0].i], f: order[0].v, iterations: history.length, history }; }
function lbfgs({ f, grad, x0, iterations = 1000, tol = 1e-8, memory = 10 }) { const n = x0.length; let x = [...x0]; let g = grad(x); const sHistory = []; const yHistory = []; const rhoHistory = []; for (let iter = 0; iter < iterations; iter++) { if (Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0)) < tol) break; const q = [...g]; const alpha = new Array(sHistory.length); for (let i = sHistory.length - 1; i >= 0; i--) { alpha[i] = rhoHistory[i] * dot(sHistory[i], q); q = q.map((qi, j) => qi - alpha[i] * yHistory[i][j]); } const gamma = sHistory.length > 0 ? dot(sHistory[sHistory.length - 1], yHistory[yHistory.length - 1]) / dot(yHistory[yHistory.length - 1], yHistory[yHistory.length - 1]) : 1 / Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0)); let r = q.map(qi => qi * gamma); for (let i = 0; i < sHistory.length; i++) { const beta = rhoHistory[i] * dot(yHistory[i], r); r = r.map((ri, j) => ri + sHistory[i][j] * (alpha[i] - beta)); } const direction = r.map(ri => -ri); let step = 1; const fx = f(x); const gDotD = dot(g, direction); let xNew = x.map((xi, j) => xi + step * direction[j]); let fNew = f(xNew); let lsIter = 0; while (fNew > fx + 0.0001 * step * gDotD && lsIter < 20) { step *= 0.5; xNew = x.map((xi, j) => xi + step * direction[j]); fNew = f(xNew); lsIter++; } const gNew = grad(xNew); const s = xNew.map((xi, j) => xi - x[j]); const y = gNew.map((gi, j) => gi - g[j]); const sy = dot(s, y); if (sy > 1e-10) { sHistory.push(s); yHistory.push(y); rhoHistory.push(1 / sy); if (sHistory.length > memory) { sHistory.shift(); yHistory.shift(); rhoHistory.shift(); } } x = xNew; g = gNew; } return { x, f: f(x), gradient: g }; }
function differentialEvolution({ f, bounds, populationSize = 50, iterations = 500, F = 0.8, CR = 0.9 }) { const n = bounds.length; const pop = []; for (let i = 0; i < populationSize; i++) { const individual = bounds.map(([lo, hi]) => lo + Math.random() * (hi - lo)); pop.push({ x: individual, f: f(individual) }); } const history = []; for (let iter = 0; iter < iterations; iter++) { for (let i = 0; i < populationSize; i++) { const candidates = []; while (candidates.length < 3) { const idx = Math.floor(Math.random() * populationSize); if (idx !== i && !candidates.includes(idx)) candidates.push(idx); } const [a, b, c] = candidates.map(idx => pop[idx].x); const mutant = a.map((ai, j) => ai + F * (b[j] - c[j])); const trial = pop[i].x.map((xi, j) => Math.random() < CR ? Math.max(bounds[j][0], Math.min(bounds[j][1], mutant[j])) : xi); const trialF = f(trial); if (trialF < pop[i].f) pop[i] = { x: trial, f: trialF }; } const best = pop.reduce((a, b) => a.f < b.f ? a : b); history.push({ iteration: iter, fBest: best.f, xBest: [...best.x] }); } const best = pop.reduce((a, b) => a.f < b.f ? a : b); return { x: best.x, f: best.f, population: pop, history }; }
function simulatedAnnealing({ f, x0, bounds = null, T0 = 1, Tmin = 1e-8, coolingRate = 0.995, iterations = 10000, stepSize = 0.1 }) { let x = [...x0]; let fx = f(x); let T = T0; let bestX = [...x]; let bestF = fx; const history = []; for (let iter = 0; iter < iterations && T > Tmin; iter++) { const xNew = x.map((xi, idx) => { let v = xi + (Math.random() - 0.5) * 2 * stepSize; if (bounds) v = Math.max(bounds[idx][0], Math.min(bounds[idx][1], v)); return v; }); const fNew = f(xNew); const delta = fNew - fx; if (delta < 0 || Math.random() < Math.exp(-delta / T)) { x = xNew; fx = fNew; if (fx < bestF) { bestF = fx; bestX = [...x]; } } T *= coolingRate; if (iter % 100 === 0) history.push({ iteration: iter, T, f: fx, bestF }); } return { x: bestX, f: bestF, history }; }
function gridSearch({ f, paramGrid }) { const keys = Object.keys(paramGrid); const values = keys.map(k => paramGrid[k]); const combinations = values.reduce((acc, vals) => { const newAcc = []; for (const combo of acc) for (const v of vals) newAcc.push([...combo, v]); return newAcc; }, [[]]); let best = { f: Infinity, params: null }; for (const combo of combinations) { const params = {}; keys.forEach((k, i) => params[k] = combo[i]); const fVal = f(params); if (fVal < best.f) best = { f: fVal, params }; } return { best, totalCombinations: combinations.length }; }
function randomSearch({ f, bounds, nTrials = 100 }) { let best = { f: Infinity, x: null }; for (let i = 0; i < nTrials; i++) { const x = bounds.map(([lo, hi]) => lo + Math.random() * (hi - lo)); const fx = f(x); if (fx < best.f) best = { f: fx, x }; } return best; }
export { gradientDescent, adam, rmsprop, nelderMead, lbfgs, differentialEvolution, simulatedAnnealing, gridSearch, randomSearch, dot };
