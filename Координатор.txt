Понятно. Твоё железо сейчас:

Материнская плата: A68HM-PLUS
Процессор: AMD Athlon 840 (4 ядра, 3.1 ГГц)
Оперативная память: 16 ГБ DDR3
Видеокарта: пока встроенная, потом GTX 1060 (6 ГБ)

Для локального координатора (не для генерации кода, а для управления проектом) этого железа более чем достаточно. Координатор — это легковесный Node.js скрипт, который читает/пишет текстовые файлы и формирует запросы. Он не требует GPU.

Ollama с DeepSeek 7B на этом железе:

Без GTX 1060: будет работать на CPU, медленно (1-2 токена/сек), но для координации — приемлемо.

С GTX 1060: будет работать быстро (15-25 токенов/сек), модель займёт ~4.7 ГБ видеопамяти.

Сейчас я дам тебе полный код координатора и инструкцию по установке Ollama с DeepSeek 7B.

Часть 1. Установка Ollama и DeepSeek 7B
Шаг 1. Установка Ollama
Открой терминал и выполни:

bash
curl -fsSL https://ollama.com/install.sh | sh
Проверь, что Ollama запустился:

bash
systemctl status ollama
Если не запущен — запусти:

bash
sudo systemctl start ollama
Шаг 2. Скачивание DeepSeek 7B (квантованной версии)
Для твоего железа (16 ГБ ОЗУ, без GPU пока) лучше всего подойдёт DeepSeek 7B в 4-битной квантизации. Она занимает ~4.7 ГБ и работает даже на CPU.

bash
ollama pull deepseek-r1:7b
Если хочешь более лёгкую модель для координации (чтобы быстрее работала на CPU):

bash
ollama pull deepseek-r1:1.5b
Рекомендую: начни с deepseek-r1:7b. Если будет тормозить — перейди на deepseek-r1:1.5b.

Шаг 3. Проверка, что модель работает
bash
ollama run deepseek-r1:7b "Привет, как дела?"
Должен появиться ответ.

#!/usr/bin/env node

/**
 * ============================================================
 * КООРДИНАТОР ПРОЕКТА CRUCIX
 * 
 * Задачи:
 * 1. Читать состояние проекта из PROJECT/STATE.txt
 * 2. Читать архитектуру из PROJECT/ARCHITECTURE.txt
 * 3. Читать текущий модуль из PROJECT/CURRENT_MODULE.txt
 * 4. Формировать запрос к локальному Ollama (DeepSeek 7B)
 * 5. Сохранять ответы
 * 6. Обновлять состояние проекта
 * 7. Делать коммиты в Git
 * 8. Готовить чистый контекст для большого ИИ
 * ============================================================
 */

import { promises as fs } from 'fs';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';
import { execSync } from 'child_process';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = join(__dirname, '..');

// ============================================================
// КОНФИГУРАЦИЯ
// ============================================================

const CONFIG = {
  PROJECT_DIR: join(ROOT, 'PROJECT'),
  CODE_DIR: join(ROOT, 'CODE'),
  DATA_DIR: join(ROOT, 'DATA'),
  LOGS_DIR: join(ROOT, 'LOGS'),
  OLLAMA_URL: 'http://localhost:11434/api/generate',
  OLLAMA_MODEL: 'deepseek-r1:7b',
};

// ============================================================
// КЛАСС КООРДИНАТОР
// ============================================================

class ProjectCoordinator {
  constructor() {
    this.state = null;
    this.architecture = null;
    this.currentModule = null;
    this.moduleDescription = null;
  }

  // ----------------------------------------------------------
  // 1. Загрузка всех файлов состояния
  // ----------------------------------------------------------

  async loadAll() {
    console.log('[Coordinator] Загрузка состояния проекта...');

    // STATE.txt
    try {
      this.state = await fs.readFile(
        join(CONFIG.PROJECT_DIR, 'STATE.txt'),
        'utf-8'
      );
    } catch (e) {
      this.state = '=== СОСТОЯНИЕ ПРОЕКТА НЕ НАЙДЕНО ===\n';
    }

    // ARCHITECTURE.txt
    try {
      this.architecture = await fs.readFile(
        join(CONFIG.PROJECT_DIR, 'ARCHITECTURE.txt'),
        'utf-8'
      );
    } catch (e) {
      this.architecture = '=== АРХИТЕКТУРА НЕ НАЙДЕНА ===\n';
    }

    // CURRENT_MODULE.txt
    try {
      this.currentModule = await fs.readFile(
        join(CONFIG.PROJECT_DIR, 'CURRENT_MODULE.txt'),
        'utf-8'
      );
    } catch (e) {
      this.currentModule = '=== ТЕКУЩИЙ МОДУЛЬ НЕ НАЙДЕН ===\n';
    }

    // MODULES/XX-название.txt
    const moduleNumber = this.extractModuleNumber(this.currentModule);
    if (moduleNumber) {
      try {
        const moduleFiles = await fs.readdir(join(CONFIG.PROJECT_DIR, 'MODULES'));
        const moduleFile = moduleFiles.find(f => f.startsWith(moduleNumber));
        if (moduleFile) {
          this.moduleDescription = await fs.readFile(
            join(CONFIG.PROJECT_DIR, 'MODULES', moduleFile),
            'utf-8'
          );
        }
      } catch (e) {
        this.moduleDescription = '=== ОПИСАНИЕ МОДУЛЯ НЕ НАЙДЕНО ===\n';
      }
    }

    console.log('[Coordinator] ✅ Состояние загружено');
  }

  // ----------------------------------------------------------
  // 2. Извлечение номера модуля из строки
  // ----------------------------------------------------------

  extractModuleNumber(text) {
    const match = text.match(/№(\d+)/);
    return match ? match[1].padStart(2, '0') : null;
  }

  // ----------------------------------------------------------
  // 3. Формирование промпта для Ollama
  // ----------------------------------------------------------

  buildPrompt() {
    return `
## ВЫ — КООРДИНАТОР ПРОЕКТА CRUCIX

### ВАША ЗАДАЧА:
Проанализировать текущее состояние проекта и предложить следующий шаг разработки.

### ТЕКУЩЕЕ СОСТОЯНИЕ ПРОЕКТА:
${this.state}

### АРХИТЕКТУРНЫЕ ПРАВИЛА:
${this.architecture}

### ТЕКУЩИЙ МОДУЛЬ:
${this.currentModule}

### ОПИСАНИЕ ТЕКУЩЕГО МОДУЛЯ:
${this.moduleDescription || 'Описание не найдено'}

### ВОПРОСЫ К ВАМ:
1. Каков статус текущего модуля? (в разработке / готов / требует доработки)
2. Что конкретно нужно сделать для завершения текущего модуля?
3. Какие файлы нужно создать или изменить?
4. Есть ли блокирующие проблемы?

### ОТВЕТЬТЕ КРАТКО, СТРУКТУРИРОВАННО:
- Статус: ...
- Следующий шаг: ...
- Файлы для создания/изменения: ...
- Проблемы: ...
`;
  }

  // ----------------------------------------------------------
  // 4. Запрос к локальному Ollama
  // ----------------------------------------------------------

  async queryLocalAI(prompt) {
    console.log('[Coordinator] Отправка запроса к локальному AI...');

    const response = await fetch(CONFIG.OLLAMA_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: CONFIG.OLLAMA_MODEL,
        prompt: prompt,
        stream: false,
        options: {
          temperature: 0.3,
          num_predict: 2000,
        },
      }),
    });

    if (!response.ok) {
      throw new Error(`Ollama ошибка: ${response.status}`);
    }

    const data = await response.json();
    console.log('[Coordinator] ✅ Ответ получен');
    return data.response;
  }

  // ----------------------------------------------------------
  // 5. Сохранение результата в лог
  // ----------------------------------------------------------

  async saveLog(analysis) {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const logFile = join(CONFIG.LOGS_DIR, `coordinator-${timestamp}.log`);
    await fs.writeFile(logFile, analysis);
    console.log(`[Coordinator] ✅ Лог сохранён: ${logFile}`);
  }

  // ----------------------------------------------------------
  // 6. Обновление STATE.txt
  // ----------------------------------------------------------

  async updateState(newState) {
    const stateFile = join(CONFIG.PROJECT_DIR, 'STATE.txt');
    await fs.writeFile(stateFile, newState);
    this.state = newState;
    console.log(`[Coordinator] ✅ STATE.txt обновлён`);
  }

  // ----------------------------------------------------------
  // 7. Git-коммит
  // ----------------------------------------------------------

  commit(message) {
    try {
      execSync('git add .', { cwd: ROOT, stdio: 'inherit' });
      execSync(`git commit -m "${message}"`, { cwd: ROOT, stdio: 'inherit' });
      console.log(`[Coordinator] ✅ Git коммит: ${message}`);
    } catch (e) {
      console.log('[Coordinator] ⚠️ Git коммит не удался (возможно, нет изменений)');
    }
  }

  // ----------------------------------------------------------
  // 8. Главный метод
  // ----------------------------------------------------------

  async run() {
    console.log('='.repeat(60));
    console.log('  КООРДИНАТОР ПРОЕКТА CRUCIX');
    console.log('='.repeat(60));

    // 1. Загружаем всё
    await this.loadAll();

    // 2. Формируем промпт
    const prompt = this.buildPrompt();

    // 3. Запрашиваем локальный AI
    try {
      const analysis = await this.queryLocalAI(prompt);
      
      // 4. Сохраняем лог
      await this.saveLog(analysis);

      // 5. Показываем результат
      console.log('\n' + '='.repeat(60));
      console.log('  АНАЛИЗ КООРДИНАТОРА');
      console.log('='.repeat(60));
      console.log(analysis);
      console.log('='.repeat(60));

      // 6. Делаем коммит
      this.commit(`Coordinator: анализ модуля ${this.extractModuleNumber(this.currentModule) || '?'}`);

      console.log('[Coordinator] ✅ Работа завершена');
    } catch (error) {
      console.error('[Coordinator] ❌ Ошибка:', error.message);
    }
  }
}

// ============================================================
// ЗАПУСК
// ============================================================

const coordinator = new ProjectCoordinator();
coordinator.run().catch(console.error);
