const $ = id => document.getElementById(id);

const STATE = {
  prediction: null,
  feedback: null,
  draft: {
    signature: null,
    comment: '',
    operatorId: 'default',
  },
};

// ========== ЗАГРУЗКА ==========

async function loadData() {
  try {
    // Прогноз
    const predRes = await fetch('../../runs/predictions/latest_forecast.json', { cache: 'no-store' });
    if (predRes.ok) STATE.prediction = await predRes.json();
  } catch (e) { console.warn('Prediction load failed:', e.message); }

  try {
    // Состояние feedback
    const fbRes = await fetch('../../runs/predictions/human_feedback.json', { cache: 'no-store' });
    if (fbRes.ok) STATE.feedback = await fbRes.json();
  } catch (e) { console.warn('Feedback load failed:', e.message); }

  // Черновик из localStorage
  const draftRaw = localStorage.getItem('crucix_hf_draft');
  if (draftRaw) {
    try {
      STATE.draft = JSON.parse(draftRaw);
      $('comment-field').value = STATE.draft.comment || '';
      $('operator-id').value = STATE.draft.operatorId || 'default';
      if (STATE.draft.signature) {
        document.querySelectorAll('.btn[data-sig]').forEach(b => {
          if (b.dataset.sig === STATE.draft.signature) b.classList.add('active');
        });
        $('save-btn').disabled = false;
      }
    } catch (e) {}
  }

  renderPrediction();
  renderMetrics();
  renderWeights();
  renderHistory();
  renderWhatIf();
  updateStats();
}

// ========== ТЕКУЩИЙ ПРОГНОЗ ==========

function renderPrediction() {
  const container = $('prediction-content');
  const meta = $('prediction-meta');
  const p = STATE.prediction;

  if (!p || !p.topRisks || p.topRisks.length === 0) {
    container.innerHTML = '<div class="empty">Прогноз не найден. Запустите pipeline или создайте заглушку runs/predictions/latest_forecast.json.</div>';
    meta.textContent = '—';
    return;
  }

  const top = p.topRisks[0];
  const ts = p.timestamp ? new Date(p.timestamp).toLocaleString('ru-RU') : '—';
  meta.textContent = ts;

  const color = top.probability > 0.7 ? 'var(--red)' :
                top.probability > 0.5 ? 'var(--orange)' :
                top.probability > 0.3 ? 'var(--yellow)' : 'var(--green)';

  container.innerHTML = `
    <div class="prediction" style="border-left-color:${color}">
      <h3>${top.name || top.id || 'Прогноз'}</h3>
      <div class="prob" style="color:${color}">${(top.probability * 100).toFixed(1)}%</div>
      <div class="details">
        Категория: ${top.category || '—'} ·
        Горизонт: ${top.horizonHours || '—'}ч ·
        Disagreement: ${top.disagreement ? (top.disagreement * 100).toFixed(1) + '%' : '—'}
      </div>
    </div>
    <div style="font-size:11px;color:var(--text-dim);margin-top:8px">
      Всего рисков в прогнозе: <strong>${p.topRisks.length}</strong>
    </div>
  `;
}

// ========== МЕТРИКИ ==========

function renderMetrics() {
  const container = $('metrics-content');
  const meta = $('metrics-meta');
  const f = STATE.feedback;

  if (!f || !f.totalFeedback) {
    container.innerHTML = '<div class="empty">Обратной связи ещё не было. Модуль ожидает сигнала оператора.</div>';
    meta.textContent = '—';
    return;
  }

  meta.textContent = f.lastUpdate ? new Date(f.lastUpdate).toLocaleString('ru-RU') : '—';

  const agreementRate = f.totalFeedback > 0
    ? ((f.goodCount + f.badCount) / f.totalFeedback * 100).toFixed(0)
    : 0;

  container.innerHTML = `
    <div style="font-family:var(--mono);font-size:12px;line-height:1.9">
      <div>Всего: <strong>${f.totalFeedback}</strong></div>
      <div>Good: <strong style="color:var(--green)">${f.goodCount}</strong></div>
      <div>Bad: <strong style="color:var(--red)">${f.badCount}</strong></div>
      <div>Uncertain: <strong style="color:var(--yellow)">${f.uncertainCount}</strong></div>
      <div>Agreement rate: <strong>${agreementRate}%</strong></div>
      <div>Признаков: <strong>${Object.keys(f.featureWeights || {}).length}</strong></div>
    </div>
    <div class="interpretation">
      <strong>Интерпретация:</strong> ${interpretMetrics(f)}
    </div>
  `;
}

function interpretMetrics(f) {
  if (!f.totalFeedback) return 'Обратной связи ещё не было.';
  if (f.totalFeedback < 5) return 'Мало данных для значимых выводов. Продолжайте размечать прогнозы.';
  const rate = (f.goodCount + f.badCount) / f.totalFeedback;
  if (rate > 0.8) return 'Высокая согласованность операторских оценок. Модель стабильна.';
  if (rate < 0.5) return 'Противоречивые оценки оператора. Возможна неоднозначность прогнозов.';
  return 'Умеренная согласованность. Продолжайте разметку.';
}

// ========== ВЕСА ==========

function renderWeights() {
  const container = $('weights-content');
  const meta = $('weights-meta');
  const f = STATE.feedback;

  if (!f || !f.featureWeights || Object.keys(f.featureWeights).length === 0) {
    container.innerHTML = '<div class="empty">Веса появятся после первого сигнала</div>';
    meta.textContent = '—';
    return;
  }

  const entries = Object.entries(f.featureWeights).sort((a, b) => b[1] - a[1]);
  const maxW = Math.max(...entries.map(([, w]) => w), 1.5);

  meta.textContent = `${entries.length} признаков`;

  container.innerHTML = entries.slice(0, 15).map(([name, w]) => {
    const pct = (w / maxW * 100).toFixed(0);
    return `
      <div class="weight-row">
        <span class="name" title="${name}">${name}</span>
        <div class="bar-wrap"><div class="bar" style="width:${pct}%"></div></div>
        <span class="num">${w.toFixed(3)}</span>
      </div>
    `;
  }).join('');
}

// ========== ИСТОРИЯ ==========

function renderHistory() {
  const container = $('history-content');
  const meta = $('history-meta');
  const f = STATE.feedback;

  if (!f || !f.feedbackHistory || f.feedbackHistory.length === 0) {
    container.innerHTML = '<div class="empty">История пуста</div>';
    meta.textContent = '0 записей';
    return;
  }

  const items = [...f.feedbackHistory].reverse().slice(0, 30);
  meta.textContent = `${f.feedbackHistory.length} записей`;

  container.innerHTML = items.map(item => {
    const ts = new Date(item.timestamp).toLocaleString('ru-RU');
    const sig = item.signature || '?';
    const comment = item.comment ? `<div class="comment-text">${escapeHtml(item.comment)}</div>` : '';
    return `
      <div class="history-item ${sig}">
        <div class="time">${ts} · ${item.operatorId || 'default'}</div>
        <span class="sig">${sig}</span>
        <span style="color:var(--text-dim);font-size:10px">${item.predictionId ? 'pred: ' + item.predictionId.slice(0, 12) : ''}</span>
        ${comment}
      </div>
    `;
  }).join('');
}

function escapeHtml(s) {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

// ========== WHAT-IF ==========

function renderWhatIf() {
  const container = $('whatif-content');
  const p = STATE.prediction;

  if (!p) {
    container.innerHTML = '<div class="empty">Нет прогноза для What-if</div>';
    return;
  }

  container.innerHTML = `
    <div style="font-size:11px;color:var(--text-dim);margin-bottom:10px">
      Упрощённый What-if: укажите значение признака, и посмотрите, как изменится прогноз.
    </div>
    <div class="row">
      <div class="field">
        <label>Признак</label>
        <select id="whatif-feature" class="comment" style="min-height:34px;margin-bottom:0">
          <option value="vix">vix</option>
          <option value="conflictCount">conflictCount</option>
          <option value="sanctionsCount">sanctionsCount</option>
          <option value="newAlerts">newAlerts</option>
          <option value="escalatedAlerts">escalatedAlerts</option>
        </select>
      </div>
      <div class="field">
        <label>Значение (0-1)</label>
        <input type="number" id="whatif-value" class="comment" style="min-height:34px;margin-bottom:0" step="0.05" min="0" max="1" value="0.8">
      </div>
    </div>
    <button class="btn" id="whatif-run" style="margin-top:8px">Рассчитать</button>
    <div id="whatif-result" style="margin-top:12px;font-size:11px;color:var(--text-dim)"></div>
  `;

  $('whatif-run').addEventListener('click', () => {
    const feature = $('whatif-feature').value;
    const value = parseFloat($('whatif-value').value);
    const baseProb = p.topRisks?.[0]?.probability || 0.5;
    const deviation = Math.abs(value - 0.5);
    const expectedShift = deviation * 0.1 * (value > 0.5 ? 1 : -1);
    const newProb = Math.max(0, Math.min(1, baseProb + expectedShift));
    $('whatif-result').innerHTML = `
      Если <strong>${feature} = ${value.toFixed(2)}</strong>:<br>
      Базовый прогноз: <strong>${(baseProb * 100).toFixed(1)}%</strong><br>
      Ожидаемый сдвиг: <strong>${(expectedShift * 100).toFixed(1)} п.п.</strong><br>
      Новый прогноз: <strong style="color:var(--accent)">${(newProb * 100).toFixed(1)}%</strong>
    `;
  });
}

// ========== СТАТИСТИКА ==========

function updateStats() {
  const f = STATE.feedback || {};
  $('stat-total').textContent = f.totalFeedback || 0;
  $('stat-good').textContent = f.goodCount || 0;
  $('stat-bad').textContent = f.badCount || 0;
  $('stat-uncertain').textContent = f.uncertainCount || 0;
}

// ========== ОБРАБОТЧИКИ КНОПОК ==========

document.querySelectorAll('.btn[data-sig]').forEach(btn => {
  btn.addEventListener('click', () => {
    document.querySelectorAll('.btn[data-sig]').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    STATE.draft.signature = btn.dataset.sig;
    $('save-btn').disabled = false;
    saveDraft();
  });
});

$('comment-field').addEventListener('input', (e) => {
  STATE.draft.comment = e.target.value;
  saveDraft();
});

$('operator-id').addEventListener('input', (e) => {
  STATE.draft.operatorId = e.target.value;
  saveDraft();
});

$('save-btn').addEventListener('click', () => {
  if (!STATE.draft.signature) {
    $('save-status').textContent = 'Сначала выберите оценку (Good/Bad/Uncertain).';
    return;
  }
  if (!STATE.prediction || !STATE.prediction.topRisks || STATE.prediction.topRisks.length === 0) {
    $('save-status').textContent = 'Нет прогноза для оценки.';
    return;
  }

  // Сохраняем в очередь черновиков
  const queue = JSON.parse(localStorage.getItem('crucix_hf_queue') || '[]');
  const top = STATE.prediction.topRisks[0];
  queue.push({
    timestamp: Date.now(),
    predictionId: `pred_${p => p.timestamp || Date.now()}`,
    signature: STATE.draft.signature,
    comment: STATE.draft.comment || null,
    operatorId: STATE.draft.operatorId || 'default',
    context: {
      features: {
        vix: STATE.prediction.fred?.vix ? Math.min(1, STATE.prediction.fred.vix / 50) : 0.4,
        conflictCount: Math.min(1, (STATE.prediction.gdelt?.conflictEvents?.length || 0) / 20),
        sanctionsCount: Math.min(1, (STATE.prediction.sanctions?.count || 0) / 10),
        newAlerts: Math.min(1, (STATE.prediction.delta?.newAlerts || 0) / 10),
        escalatedAlerts: Math.min(1, (STATE.prediction.delta?.escalatedAlerts || 0) / 5),
      },
      predictedProb: top.probability,
      actualOutcome: null,
    },
  });
  localStorage.setItem('crucix_hf_queue', JSON.stringify(queue));

  // Сброс формы
  document.querySelectorAll('.btn[data-sig]').forEach(b => b.classList.remove('active'));
  STATE.draft.signature = null;
  $('save-btn').disabled = true;
  $('save-status').innerHTML = `✓ Метка добавлена в очередь. Всего в очереди: <strong>${queue.length}</strong>. Не забудьте экспортировать и применить.`;
});

$('export-btn').addEventListener('click', () => {
  const queue = JSON.parse(localStorage.getItem('crucix_hf_queue') || '[]');
  if (queue.length === 0) {
    $('save-status').textContent = 'Очередь пуста. Нечего экспортировать.';
    return;
  }
  const blob = new Blob([JSON.stringify(queue, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);

# План встраивания Human Feedback в engine.mjs

**Статус: план. Патч пока не применён.**

Этот документ — инструкция для того момента, когда будет решено подключить `human_feedback.mjs` к основному конвейеру Crucix.

---

## Куда встраивать

Рекомендованное место — **после фазы F (калибровка), перед фазой G (активное обучение)**.

Причина. Фаза F обновляет Brier Score и Platt-калибровку. Если подключить human_feedback раньше — калибровка срежет его эффект. Если позже — обновлённые веса не попадут в текущий прогноз. Оптимальная точка: сразу после калибровки, до активного обучения.

Схема конвейера с новой фазой:

~~~
sweep (15 мин)
  │
  ├─ Фаза A: источники
  ├─ Фаза B: базовые модели
  ├─ Фаза C: продвинутые модели
  ├─ Фаза D: ансамблирование
  ├─ Фаза E: рефлексивная коррекция
  ├─ Фаза F: калибровка
  ├─ Фаза F': human feedback ← НОВОЕ
  ├─ Фаза G: активное обучение
  ├─ Фаза H: объяснимость
  ├─ Фаза I: граф знаний
  │
  └─ latest_forecast.json → дашборд
~~~

---

## Патч engine.mjs — минимальный

В `apis/predict/engine.mjs` после блока фазы F (примерно после строки с `phaseF_Calibration`) добавить импорт:

~~~javascript
import { crucixHumanFeedback } from '../features/human-feedback/human_feedback.mjs';
~~~

**Внимание:** путь `../features/...` — из `apis/predict/` в `features/human-feedback/`. Проверить корректность после переноса в рабочий проект.

Затем в главной функции `runForecastPipeline`, сразу после вызова `phaseF_Calibration`:

~~~javascript
// ─── Фаза F': Human Feedback ───
const humanFeedback = options.skipHumanFeedback
  ? null
  : crucixHumanFeedback(latest, calibratedForecasts, {
      save: true,
      applyNewWeights: false,  // пока веса не используются потребителем
    });
~~~

И в финальный объект `forecast` добавить поле:

~~~javascript
forecast: {
  // ... существующие поля ...
  humanFeedback,
  // ...
}
~~~

Плюс новая опция в `options`:

~~~javascript
const {
  skipSources = false,
  skipAdvanced = false,
  skipExplainability = false,
  skipActiveLearning = false,
  skipKnowledgeGraph = false,
  skipHumanFeedback = false,  // ← новая опция
} = options;
~~~

## Патч engine.mjs — расширенный (с применением весов)

Этот вариант нужен, когда появится потребитель весов — модуль, который читает обновлённые веса и использует их в ансамблировании.

В `phaseD_Ensemble` перед расчётом финальной вероятности:

~~~javascript
import { HumanFeedbackLearner } from '../features/human-feedback/human_feedback.mjs';

// Загружаем обученные веса признаков
const humanLearner = HumanFeedbackLearner.load();
const featureWeights = humanLearner.getCurrentWeights();

// Применяем в ансамблировании
const estimate = {
  source: 'bayesian',
  probability: bayesF.posterior,
  weight: 1.0 * (featureWeights.vix || 1.0),
};
~~~

Это сложнее, потому что требует определить, какие именно признаки и как влияют на вес источника. На первом этапе — не делать, начать с минимального патча.

Чтобы применить эти метки к состоянию обучения, нужно:

### Шаг 1. Сохранить скачанный файл

Оператор скачивает файл из браузера. Его надо переместить в удобное место:

~~~bash

~~~

### Шаг 2. Применить через скрипт

Готовый скрипт для применения:

~~~bash
cd "/home/ta8/Документы" && node --input-type=module -e "

  }
}

console.log('Применено меток:', applied);
console.log('Всего обратной связи:', learner.totalFeedback);

console.log('Uncertain:', learner.uncertainCount);
"
~~~

## Что требует доработки

Пять мест, которые сейчас работают неоптимально.

### 1. Нет серверного эндпоинта для метки

Дашборд не может писать метку напрямую в файл. Оператор скачивает JSON и применяет командой в терминале. Это работает, но неудобно.

**Решение:** добавить эндпоинт `POST /api/human_feedback/record` в основной сервер Crucix. Тогда дашборд сможет отправлять метки напрямую.

Это отдельная задача для основной серверной части (`server/router.mjs`, `apis/sources/human_feedback-api.mjs`), не для этой папки.

### 2. Потребитель весов отсутствует

Обновлённые веса сохраняются в `human_feedback.json`, но никто их не читает. Нужен патч в `phaseD_Ensemble` — либо в ансамблировании, либо в отдельном шаге после human_feedback, который применяет веса к текущим прогнозам.

**Решение:** патч в `apis/predict/engine.mjs` (см. расширенный вариант выше).

### 3. Текстовые комментарии не обрабатываются

Сейчас комментарии сохраняются в истории, но не влияют на обучение. Их можно:
- Оставить как есть (для контекста аналитика при следующем просмотре).
- Отправлять в LLM для извлечения дополнительных признаков.
- Использовать для дообучения LLM-слоя.

**Решение для первого этапа:** оставить как есть. Обработка через LLM — отдельная задача.

### 4. Калибровка весов между срезами времени

Пока нет механизма затухания: если оператор дал метку 3 месяца назад, она весит столько же, сколько метка сегодня. Нужно ввести временной распад (например, экспоненциальный с half-life 30 дней).

**Решение:** параметр `recencyDecay` в `HumanFeedbackLearner` (не реализован).

### 5. Нет тестов

Модуль `human_feedback.mjs` не покрыт тестами. Нужны:
- Unit-тесты для `recordFeedback`, `updateWeights`, `whatIf`.
- Property-based тест: сумма весов не уходит в бесконечность при большом количестве меток.
- Fuzz-тест: корректная обработка невалидных входных данных.

**Решение:** файл `tests/predict/human_feedback.test.mjs` (не создан).

## Рекомендации по порядку внедрения

Если бы это делалось с нуля, порядок был бы такой.

### Этап 1. Только сохранить модуль, не подключать

Сейчас модуль лежит в `features/human-feedback/` как заготовка. Ничего не вызывает. Это безопасное состояние.

### Этап 2. Подключить минимальный патч

Добавить в `engine.mjs`:
- Импорт `crucixHumanFeedback`.
- Опцию `skipHumanFeedback`.
- Вызов после фазы F (без применения весов).

Эффект: в `latest_forecast.json` появится поле `humanFeedback` с метриками (но не с весами, применёнными к прогнозу). Наблюдаем.

### Этап 3. Включить применение весов

Доработать `phaseD_Ensemble`, чтобы он читал веса из `human_feedback.json` и применял их.

Эффект: обратная связь оператора начнёт влиять на прогнозы. Это уже реальный обучающий цикл.

### Этап 4. Серверный эндпоинт

Добавить `POST /api/human_feedback/record`, обновить дашборд. Оператор сможет ставить метки прямо в браузере, они сразу будут попадать в файл.

### Этап 5. Тесты и регрессионный мониторинг

Покрыть тестами, добавить в CI, мониторить деградацию весов.
