function renderSignalRow(s) {
  const severityClass = s.severity === 'critical' ? 'critical' : '';
  const trendHtml = s.trend
    ? '<span class="adv-signal-trend ' + s.trend + '">' + (s.trend === 'up' ? '↑' : '↓') + '</span>'
    : '';
  const escapedTooltip = String(s.tooltip || '')
    .replace(/"/g, '&quot;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;');
  return (
    '<div class="adv-signal ' + severityClass + '" style="border-left-color:' + s.color + ';" data-tooltip="' + escapedTooltip + '">' +
    '<span class="adv-signal-icon">' + s.icon + '</span>' +
    '<div class="adv-signal-content">' +
    '<div class="adv-signal-label">' + s.label + '</div>' +
    '<div class="adv-signal-value" style="color:' + s.color + ';">' + s.value + trendHtml + '</div>' +
    '</div></div>'
  );
}

function renderAdvancedPreview(data) {
  const el = $('advanced-preview-content');
  const footer = $('advanced-footer');
  if (!el) return;

  const signals = [];
  for (const def of ADVANCED_SIGNAL_DEFS) {
    try {
      const extracted = def.extract(data);
      if (extracted) signals.push({ ...def, ...extracted });
    } catch (e) {
      console.warn('[advanced-preview] Signal ' + def.id + ' extraction failed:', e.message);
    }
  }

  const sections = {};
  for (const sig of signals) {
    if (!sections[sig.section]) sections[sig.section] = [];
    sections[sig.section].push(sig);
  }

  if (signals.length === 0) {
    el.innerHTML =
      '<div class="adv-empty">' +
      'Advanced-модули неактивны в последнем прогнозе.<br>' +
      'Запустите pipeline с опциями по умолчанию.' +
      '<a href="/advanced.html">Открыть Advanced Analytics →</a>' +
      '</div>';
    if (footer) footer.style.display = 'none';
    return;
  }

  const html = Object.entries(sections).map(([sectionName, sigs]) =>
    '<div class="adv-section-label">' + sectionName + '</div>' +
    sigs.map(s => renderSignalRow(s)).join('')
  ).join('');
  el.innerHTML = html;

  const criticalCount = signals.filter(s => s.severity === 'critical' || s.severity === 'high').length;
  if (footer) {
    footer.style.display = 'flex';
    const dotEl = $('footer-status-dot');
    if (dotEl) {
      dotEl.className = 'adv-footer-dot' + (criticalCount > 2 ? ' critical' : criticalCount > 0 ? ' warning' : '');
    }
    const statusEl = $('footer-status-text');
    if (statusEl) statusEl.textContent = criticalCount > 0 ? criticalCount + ' критичных' : 'все стабильно';
    const refreshEl = $('footer-refresh');
    if (refreshEl && data.timestamp) {
      refreshEl.textContent = new Date(data.timestamp).toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
    }
  }

  const badge = $('badge-advanced');
  if (badge) {
    if (criticalCount > 0) {
      badge.textContent = criticalCount > 9 ? '9+' : String(criticalCount);
      badge.style.display = 'flex';
    } else {
      badge.style.display = 'none';
    }
  }
}

async function load() {
  try {
    const res = await fetch('../runs/predictions/latest_forecast.json');
    if (!res.ok) throw new Error('HTTP ' + res.status);
    const data = await res.json();
    render(data);
  } catch (e) {
    document.querySelectorAll('.empty').forEach(el => el.textContent = 'Нет данных: ' + e.message);
  }
}

function render(data) {
  if (data.timestamp) $('last-update').textContent = new Date(data.timestamp).toLocaleTimeString('ru-RU');
  if (data.sweepId) $('sweep-id').textContent = 'sweep: ' + data.sweepId.substring(0, 12);

  // Top risks
  if (data.topRisks && data.topRisks.length > 0) {
    $('top-risks').innerHTML = data.topRisks.map(r => {
      const color = COLORS[r.level || 'moderate'] || '#3b82f6';
      const prob = r.probability || 0;
      return '<div class="event-row" style="border-left-color:' + color + '">' +
        '<div class="event-header"><span class="event-name">' + (r.name || r.id) + '</span>' +
        '<span class="event-prob" style="color:' + color + '">' + (prob * 100).toFixed(1) + '%</span></div>' +
        '<div class="bar"><div class="bar-fill" style="width:' + (prob * 100) + '%;background:' + color + '"></div></div>' +
        '</div>';
    }).join('');
  }

  // System state
  if (data.systemState) {
    const s = data.systemState;
    $('system-state').innerHTML = '<div style="font-size:20px;font-weight:700;color:var(--accent);margin-bottom:8px">' + (s.current || '—') + '</div>' +
      '<div style="font-size:11px;color:var(--text-dim);font-family:var(--mono)">Markov next: ' + (s.markov?.next || '—') + '</div>';
  }

  // Brier
  if (data.calibration) {
    const brier = data.calibration.brierScores || {};
    const entries = Object.entries(brier).sort((a, b) => (a[1].score || 0) - (b[1].score || 0));
    if (entries.length > 0) {
      $('brier-info').innerHTML = entries.map(([src, b]) => {
        const score = b.score || 0;
        const color = score < 0.1 ? '#22c55e' : score < 0.2 ? '#eab308' : '#ef4444';
        return '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid var(--border);font-size:12px">' +
          '<span>' + src + '</span>' +
          '<span style="font-family:var(--mono);color:' + color + ';font-weight:700">' + score.toFixed(4) + '</span></div>';
      }).join('');
    }
  }

  // Advanced preview
  renderAdvancedPreview(data);
}

setInterval(load, 60000);
