function linearBetaSchedule(T = 1000, betaStart = 1e-4, betaEnd = 0.02) { const betas = []; for (let t = 0; t < T; t++) betas.push(betaStart + (betaEnd - betaStart) * (t / (T - 1))); const alphas = betas.map(b => 1 - b); const alphaBars = []; let prod = 1; for (const a of alphas) { prod *= a; alphaBars.push(prod); } return { betas, alphas, alphaBars }; }
function cosineBetaSchedule(T = 1000, s = 0.008) { const betas = []; const alphaBars = []; const f = t => Math.cos(((t / T + s) / (1 + s)) * Math.PI * 0.5) ** 2; const f0 = f(0); for (let t = 0; t < T; t++) alphaBars.push(f(t) / f0); for (let t = 0; t < T; t++) { if (t === 0) betas.push(1 - alphaBars[0]); else betas.push(Math.min(0.999, 1 - alphaBars[t] / alphaBars[t - 1])); } return { betas, alphaBars, alphas: betas.map(b => 1 - b) }; }
function xavier(rows, cols, seed) { const scale = Math.sqrt(2 / (rows + cols)); const W = []; let s = seed || 42; const rand = () => { s = (s * 9301 + 49297) % 233280; return s / 233280; }; for (let i = 0; i < cols; i++) { const col = new Float64Array(rows); for (let j = 0; j < rows; j++) { const u1 = rand(), u2 = rand(); const z = Math.sqrt(-2 * Math.log(u1 + 1e-10)) * Math.cos(2 * Math.PI * u2); col[j] = z * scale; } W.push(col); } return W; }
function timeEmbedding(t, dim) { const emb = new Float64Array(dim); for (let i = 0; i < dim; i += 2) { const freq = Math.exp(-Math.log(10000) * i / dim); emb[i] = Math.sin(t * freq); if (i + 1 < dim) emb[i + 1] = Math.cos(t * freq); } return emb; }
function gaussianSample() { const u1 = Math.random(), u2 = Math.random(); return Math.sqrt(-2 * Math.log(u1 + 1e-10)) * Math.cos(2 * Math.PI * u2); }

class DenoiseNet {
  constructor({ inputDim = 10, timeDim = 32, hiddenDim = 128, nLayers = 3, seed = 42 }) { this.inputDim = inputDim; this.timeDim = timeDim; this.hiddenDim = hiddenDim; this.nLayers = nLayers; const layers = []; let prevDim = inputDim + timeDim; for (let i = 0; i < nLayers; i++) { layers.push({ W: xavier(prevDim, hiddenDim, seed + i * 10), b: new Float64Array(hiddenDim), inDim: prevDim, outDim: hiddenDim }); prevDim = hiddenDim; } layers.push({ W: xavier(prevDim, inputDim, seed + 999), b: new Float64Array(inputDim), inDim: prevDim, outDim: inputDim }); this.layers = layers; }
  forward(x_t, t) { const tEmb = timeEmbedding(t, this.timeDim); let h = new Float64Array(this.inputDim + this.timeDim); for (let i = 0; i < this.inputDim; i++) h[i] = x_t[i]; for (let i = 0; i < this.timeDim; i++) h[this.inputDim + i] = tEmb[i]; const activations = [h]; for (let i = 0; i < this.layers.length; i++) { const layer = this.layers[i]; const out = new Float64Array(layer.outDim); for (let o = 0; o < layer.outDim; o++) { let sum = layer.b[o]; for (let j = 0; j < layer.inDim; j++) sum += h[j] * layer.W[o][j]; out[o] = sum; } if (i < this.layers.length - 1) for (let o = 0; o < layer.outDim; o++) out[o] = Math.max(0, out[o]); h = out; activations.push(h); } return { output: h, activations }; }
  backward(dOutput, activations) { const grads = this.layers.map(layer => ({ dW: layer.W.map(row => new Float64Array(row.length)), db: new Float64Array(layer.outDim) })); let dH = dOutput; for (let l = this.layers.length - 1; l >= 0; l--) { const layer = this.layers[l]; const input = activations[l]; for (let o = 0; o < layer.outDim; o++) { for (let j = 0; j < layer.inDim; j++) grads[l].dW[o][j] = dH[o] * input[j]; grads[l].db[o] = dH[o]; } const dInput = new Float64Array(layer.inDim); for (let j = 0; j < layer.inDim; j++) { let sum = 0; for (let o = 0; o < layer.outDim; o++) sum += dH[o] * layer.W[o][j]; dInput[j] = sum; } if (l < this.layers.length - 1 && l > 0) { const prevActivation = activations[l]; for (let j = 0; j < dInput.length; j++) if (prevActivation[j] <= 0) dInput[j] = 0; } dH = dInput; } return grads; }
  update(grads, lr = 0.001) { for (let l = 0; l < this.layers.length; l++) { const layer = this.layers[l]; for (let o = 0; o < layer.outDim; o++) { for (let j = 0; j < layer.inDim; j++) layer.W[o][j] -= lr * grads[l].dW[o][j]; layer.b[o] -= lr * grads[l].db[o]; } } }
}

class DDPM {
  constructor({ dataDim = 10, T = 200, hiddenDim = 128, nLayers = 3, schedule = 'linear', seed = 42 }) { this.dataDim = dataDim; this.T = T; const { betas, alphas, alphaBars } = schedule === 'cosine' ? cosineBetaSchedule(T) : linearBetaSchedule(T); this.betas = betas; this.alphas = alphas; this.alphaBars = alphaBars; this.sigmas = betas.map(b => Math.sqrt(b)); this.net = new DenoiseNet({ inputDim: dataDim, timeDim: 32, hiddenDim, nLayers, seed }); }
  q_sample(x0, t, noise = null) { const alphaBar = this.alphaBars[t]; const sqrtAlphaBar = Math.sqrt(alphaBar); const sqrtOneMinusAlphaBar = Math.sqrt(1 - alphaBar); const eps = noise || Array.from({ length: this.dataDim }, () => gaussianSample()); const x_t = new Float64Array(this.dataDim); for (let i = 0; i < this.dataDim; i++) x_t[i] = sqrtAlphaBar * x0[i] + sqrtOneMinusAlphaBar * eps[i]; return { x_t, noise: eps }; }
  p_sample(x_t, t) { const { output: predictedNoise } = this.net.forward(x_t, t); const alpha = this.alphas[t]; const alphaBar = this.alphaBars[t]; const beta = this.betas[t]; const sqrtAlpha = Math.sqrt(alpha); const sqrtOneMinusAlphaBar = Math.sqrt(1 - alphaBar); const mu = new Float64Array(this.dataDim); for (let i = 0; i < this.dataDim; i++) mu[i] = (1 / sqrtAlpha) * (x_t[i] - (beta / sqrtOneMinusAlphaBar) * predictedNoise[i]); if (t > 0) { const sigma = this.sigmas[t]; const z = Array.from({ length: this.dataDim }, () => gaussianSample()); const x_prev = new Float64Array(this.dataDim); for (let i = 0; i < this.dataDim; i++) x_prev[i] = mu[i] + sigma * z[i]; return { x_prev, predictedNoise, mu }; } return { x_prev: mu, predictedNoise, mu }; }
  generate({ startFrom = null, returnTrajectory = false } = {}) { let x = startFrom || Array.from({ length: this.dataDim }, () => gaussianSample()); const trajectory = returnTrajectory ? [{ t: this.T, x: [...x] }] : []; for (let t = this.T - 1; t >= 0; t--) { const { x_prev } = this.p_sample(x, t); x = Array.from(x_prev); if (returnTrajectory && t % 10 === 0) trajectory.push({ t, x: [...x] }); } return { sample: x, trajectory }; }
  train_step(x0, lr = 0.001) { const t = Math.floor(Math.random() * this.T); const { x_t, noise } = this.q_sample(x0, t); const { output: predictedNoise, activations } = this.net.forward(x_t, t); let loss = 0; const dOutput = new Float64Array(this.dataDim); for (let i = 0; i < this.dataDim; i++) { const diff = predictedNoise[i] - noise[i]; loss += diff * diff; dOutput[i] = 2 * diff / this.dataDim; } loss /= this.dataDim; const grads = this.net.backward(dOutput, activations); this.net.update(grads, lr); return { loss, t }; }
  fit(X, { epochs = 100, batchSize = 16, lr = 0.001, verbose = false } = {}) { const n = X.length; const history = []; for (let epoch = 0; epoch < epochs; epoch++) { const indices = Array.from({ length: n }, (_, i) => i); for (let i = n - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [indices[i], indices[j]] = [indices[j], indices[i]]; } let epochLoss = 0, count = 0; for (let start = 0; start < n; start += batchSize) { const batch = indices.slice(start, start + batchSize); for (const idx of batch) { const { loss } = this.train_step(X[idx], lr); epochLoss += loss; count++; } } const avgLoss = epochLoss / count; history.push({ epoch, loss: avgLoss }); if (verbose && epoch % 10 === 0) console.log(`Epoch ${epoch}, loss: ${avgLoss.toFixed(6)}`); } return { history, finalLoss: history[history.length - 1].loss }; }
}
