async function load() {
  try {
    const res = await fetch(CONFIG.dataUrl + '?t=' + Date.now(), { cache: 'no-store' });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    STATE.data = await res.json();
    STATE.lastUpdate = new Date();
    render();
    addLiveEvent({
      type: 'prediction',
      message: `Обновлён прогноз · composite=${STATE.data.compositeRisk?.composite?.toFixed(3) || '—'}`,
    });
  } catch (e) {
    console.error('[cockpit] Load error:', e);
    renderError(e.message);
  }
}

  const cr = d.compositeRisk;
  const banner = $('alert-banner');
  if (cr?.level === 'critical' || cr?.level === 'high') {
    banner.classList.add('visible');
    $('alert-banner-title').textContent = `Уровень риска: ${cr.level.toUpperCase()}`;
    const top = cr.topDrivers?.[0];
    $('alert-banner-desc').textContent = top
      ? `${top.name}: ${(top.value * 100).toFixed(1)}% — ${top.rationale || ''}`
      : `Composite risk: ${cr.composite.toFixed(3)}`;
  } else {
    banner.classList.remove('visible');
  }
}

function addLiveEvent(event) {
  STATE.liveEvents.unshift({ ...event, timestamp: new Date() });
  if (STATE.liveEvents.length > STATE.maxLiveEvents) {
    STATE.liveEvents = STATE.liveEvents.slice(0, STATE.maxLiveEvents);
  }
  renderLiveFeed();
}

function renderLiveFeed() {
  const container = $('live-feed');
  $('live-feed-count').textContent = `${STATE.liveEvents.length} событий`;
  if (STATE.liveEvents.length === 0) {
    container.innerHTML = '<div class="empty">Ожидание событий...</div>';
    return;
  }
  container.innerHTML = STATE.liveEvents.slice(0, 15).map(e => `
    <div class="live-feed-item ${e.type || ''}">
      <div class="live-feed-time">${e.timestamp.toLocaleTimeString('ru-RU')}</div>
      <div>${e.message}</div>
    </div>
  `).join('');
}

async function runPipeline() {
  addLiveEvent({ type: 'signal', message: 'Запуск pipeline вручную...' });
  try {
    const res = await fetch('/api/pipeline/run', { method: 'POST' });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    addLiveEvent({ type: 'prediction', message: 'Pipeline запущен' });
    setTimeout(load, 5000);
  } catch (e) {
    addLiveEvent({ type: 'alert', message: `Ошибка запуска: ${e.message}` });
  }
}

function toggleAutoRefresh() {
  CONFIG.autoRefresh = !CONFIG.autoRefresh;
  $('refresh-label').textContent = `Auto: ${CONFIG.autoRefresh ? 'ON' : 'OFF'}`;
  if (CONFIG.autoRefresh) {
    startAutoRefresh();
    addLiveEvent({ type: 'signal', message: 'Auto-refresh включён' });
  } else {
    stopAutoRefresh();
    addLiveEvent({ type: 'signal', message: 'Auto-refresh выключен' });
  }
}

function startAutoRefresh() {
  stopAutoRefresh();
  STATE.autoRefreshTimer = setInterval(load, CONFIG.refreshMs);
}

function stopAutoRefresh() {
  if (STATE.autoRefreshTimer) {
    clearInterval(STATE.autoRefreshTimer);
    STATE.autoRefreshTimer = null;
  }
}

function emergencyStop() {
  if (!confirm('Emergency stop: остановить auto-refresh и все фоновые процессы?')) return;
  stopAutoRefresh();
  CONFIG.autoRefresh = false;
  addLiveEvent({ type: 'alert', message: 'EMERGENCY STOP' });
  $('refresh-label').textContent = 'Auto: OFF';
}

function downloadJSON() {
  if (!STATE.data) return;
  const blob = new Blob([JSON.stringify(STATE.data, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `crucix_forecast_${Date.now()}.json`;
  a.click();
  URL.revokeObjectURL(url);
  addLiveEvent({ type: 'signal', message: 'Прогноз скачан' });
}

let paletteSelection = 0;
let filteredCommands = [...COMMANDS];

function openCommandPalette() {
  $('command-palette').classList.add('active');
  $('command-input').value = '';
  $('command-input').focus();
  paletteSelection = 0;
  filterCommands('');
}

function closeCommandPalette() {
  $('command-palette').classList.remove('active');
}

function filterCommands(query) {
  const q = query.toLowerCase().trim();
  filteredCommands = q
    ? COMMANDS.filter(c => c.label.toLowerCase().includes(q) || c.id.toLowerCase().includes(q))
    : COMMANDS;
  paletteSelection = 0;
  renderCommandList();
}

function renderCommandList() {
  const container = $('command-list');
  if (filteredCommands.length === 0) {
    container.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--text-faint);">Ничего не найдено</div>';
    return;
  }
  container.innerHTML = filteredCommands.map((cmd, i) => `
    <div class="command-item ${i === paletteSelection ? 'selected' : ''}" onclick="executeCommand(${i})">
      <span class="command-icon">${cmd.icon}</span>
      <span class="command-label">${cmd.label}</span>
      ${cmd.shortcut ? `<span class="command-shortcut">${cmd.shortcut}</span>` : ''}
    </div>
  `).join('');
}

function executeCommand(index) {
  const cmd = filteredCommands[index];
  if (!cmd) return;
  closeCommandPalette();
  cmd.action();
}

document.addEventListener('keydown', (e) => {
  if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
    e.preventDefault();
    if ($('command-palette').classList.contains('active')) closeCommandPalette();
    else openCommandPalette();
    return;
  }
  if ($('command-palette').classList.contains('active')) {
    if (e.key === 'Escape') { closeCommandPalette(); return; }
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      paletteSelection = Math.min(paletteSelection + 1, filteredCommands.length - 1);
      renderCommandList();
      return;
    }
    if (e.key === 'ArrowUp') {
      e.preventDefault();
      paletteSelection = Math.max(paletteSelection - 1, 0);
      renderCommandList();
      return;
    }
    if (e.key === 'Enter') {
      e.preventDefault();
      executeCommand(paletteSelection);
      return;
    }
    return;
  }
  if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
  if (e.key === 'r' && !e.ctrlKey && !e.metaKey) runPipeline();
  if (e.key === 'a' && !e.ctrlKey && !e.metaKey) toggleAutoRefresh();
  if (e.key === 'e' && !e.ctrlKey && !e.metaKey) downloadJSON();
});

let nextCycle = Date.now() + 900000;

function updateTimer() {
  const remaining = Math.max(0, nextCycle - Date.now());
  const mins = Math.floor(remaining / 60000);
  const secs = Math.floor((remaining % 60000) / 1000);
  $('timer-value').textContent = `${mins}:${String(secs).padStart(2, '0')}`;
  if (remaining <= 0) nextCycle = Date.now() + 900000;
}

$('command-input').addEventListener('input', (e) => filterCommands(e.target.value));

load();
startAutoRefresh();
setInterval(updateTimer, 1000);

addLiveEvent({ type: 'signal', message: 'Cockpit v5.0 запущен' });
